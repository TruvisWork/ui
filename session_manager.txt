import time
import uuid
from typing import Dict, Optional, Set
from datetime import datetime, timedelta
import asyncio
import threading
from dataclasses import dataclass
import logging

logger = logging.getLogger(__name__)

@dataclass
class UserSession:
    session_id: str
    username: str
    uid: str
    markets: list
    login_time: datetime
    last_activity: datetime
    ip_address: str
    user_agent: str
    is_active: bool = True

class SessionManager:
    """
    Manages user sessions with the following features:
    - Single session per user (kicks out other sessions)
    - Automatic session timeout based on inactivity
    - Session cleanup
    - Session validation
    """
    
    def __init__(self, 
                 session_timeout_minutes: int = 30,
                 max_session_lifetime_hours: int = 24,
                 cleanup_interval_minutes: int = 5):
        
        # Session storage: {session_id: UserSession}
        self.sessions: Dict[str, UserSession] = {}
        
        # User to session mapping: {username: session_id}
        self.user_sessions: Dict[str, str] = {}
        
        # Configuration
        self.session_timeout = timedelta(minutes=session_timeout_minutes)
        self.max_session_lifetime = timedelta(hours=max_session_lifetime_hours)
        self.cleanup_interval = timedelta(minutes=cleanup_interval_minutes)
        
        # Start cleanup task
        self._cleanup_task = None
        self.start_cleanup_task()
        
        logger.info("SessionManager initialized - Timeout: %dm, Max lifetime: %dh, Cleanup: %dm",
                   session_timeout_minutes, max_session_lifetime_hours, cleanup_interval_minutes)
    
    def start_cleanup_task(self):
        """Start the background cleanup task"""
        def cleanup_loop():
            while True:
                try:
                    self.cleanup_expired_sessions()
                    time.sleep(self.cleanup_interval.total_seconds())
                except Exception as e:
                    logger.error("Session cleanup error: %s", str(e))
                    time.sleep(60)  # Wait 1 minute on error
        
        self._cleanup_task = threading.Thread(target=cleanup_loop, daemon=True)
        self._cleanup_task.start()
        logger.info("Session cleanup task started")
    
    def create_session(self, username: str, uid: str, markets: list, 
                      ip_address: str, user_agent: str) -> UserSession:
        """
        Create a new session for a user.
        If user already has an active session, it will be terminated.
        """
        # Check if user already has an active session
        self._terminate_user_sessions(username)
        
        # Create new session
        session_id = str(uuid.uuid4())
        now = datetime.utcnow()
        
        session = UserSession(
            session_id=session_id,
            username=username,
            uid=uid,
            markets=markets,
            login_time=now,
            last_activity=now,
            ip_address=ip_address,
            user_agent=user_agent,
            is_active=True
        )
        
        # Store session
        self.sessions[session_id] = session
        self.user_sessions[username] = session_id
        
        logger.info("SESSION_CREATED - User: %s, SessionID: %s, IP: %s", 
                   username, session_id[:8], ip_address)
        
        return session
    
    def validate_session(self, session_id: str, update_activity: bool = True) -> Optional[UserSession]:
        """
        Validate a session and optionally update last activity.
        Returns None if session is invalid or expired.
        """
        if not session_id:
            return None
        
        session = self.sessions.get(session_id)
        if not session:
            logger.warning("SESSION_NOT_FOUND - SessionID: %s", session_id[:8])
            return None
        
        if not session.is_active:
            logger.warning("SESSION_INACTIVE - User: %s, SessionID: %s", 
                          session.username, session_id[:8])
            return None
        
        now = datetime.utcnow()
        
        # Check if session has expired due to inactivity
        if now - session.last_activity > self.session_timeout:
            logger.info("SESSION_EXPIRED_INACTIVITY - User: %s, SessionID: %s, Inactive for: %s",
                       session.username, session_id[:8], now - session.last_activity)
            self._terminate_session(session_id)
            return None
        
        # Check if session has exceeded maximum lifetime
        if now - session.login_time > self.max_session_lifetime:
            logger.info("SESSION_EXPIRED_LIFETIME - User: %s, SessionID: %s, Lifetime: %s",
                       session.username, session_id[:8], now - session.login_time)
            self._terminate_session(session_id)
            return None
        
        # Update last activity if requested
        if update_activity:
            session.last_activity = now
            logger.debug("SESSION_ACTIVITY_UPDATED - User: %s, SessionID: %s", 
                        session.username, session_id[:8])
        
        return session
    
    def terminate_session(self, session_id: str) -> bool:
        """Manually terminate a specific session"""
        return self._terminate_session(session_id)
    
    def terminate_user_sessions(self, username: str) -> int:
        """Terminate all sessions for a specific user"""
        return self._terminate_user_sessions(username)
    
    def _terminate_session(self, session_id: str) -> bool:
        """Internal method to terminate a session"""
        session = self.sessions.get(session_id)
        if not session:
            return False
        
        # Mark as inactive
        session.is_active = False
        
        # Remove from user sessions mapping
        if session.username in self.user_sessions:
            if self.user_sessions[session.username] == session_id:
                del self.user_sessions[session.username]
        
        # Remove from sessions storage
        del self.sessions[session_id]
        
        logger.info("SESSION_TERMINATED - User: %s, SessionID: %s", 
                   session.username, session_id[:8])
        return True
    
    def _terminate_user_sessions(self, username: str) -> int:
        """Internal method to terminate all sessions for a user"""
        terminated_count = 0
        
        # Find all sessions for the user
        sessions_to_terminate = []
        for session_id, session in self.sessions.items():
            if session.username == username and session.is_active:
                sessions_to_terminate.append(session_id)
        
        # Terminate found sessions
        for session_id in sessions_to_terminate:
            if self._terminate_session(session_id):
                terminated_count += 1
        
        if terminated_count > 0:
            logger.info("USER_SESSIONS_TERMINATED - User: %s, Count: %d", 
                       username, terminated_count)
        
        return terminated_count
    
    def cleanup_expired_sessions(self):
        """Remove expired sessions"""
        now = datetime.utcnow()
        expired_sessions = []
        
        for session_id, session in self.sessions.items():
            # Check for inactivity timeout
            if now - session.last_activity > self.session_timeout:
                expired_sessions.append(session_id)
            # Check for maximum lifetime
            elif now - session.login_time > self.max_session_lifetime:
                expired_sessions.append(session_id)
        
        cleaned_count = 0
        for session_id in expired_sessions:
            if self._terminate_session(session_id):
                cleaned_count += 1
        
        if cleaned_count > 0:
            logger.info("SESSION_CLEANUP - Expired sessions removed: %d", cleaned_count)
        
        logger.debug("SESSION_CLEANUP - Active sessions: %d", len(self.sessions))
    
    def get_session_info(self, session_id: str) -> Optional[Dict]:
        """Get session information for debugging/monitoring"""
        session = self.sessions.get(session_id)
        if not session:
            return None
        
        return {
            "session_id": session.session_id,
            "username": session.username,
            "uid": session.uid,
            "markets": session.markets,
            "login_time": session.login_time.isoformat(),
            "last_activity": session.last_activity.isoformat(),
            "ip_address": session.ip_address,
            "user_agent": session.user_agent,
            "is_active": session.is_active,
            "time_since_login": str(datetime.utcnow() - session.login_time),
            "time_since_activity": str(datetime.utcnow() - session.last_activity)
        }
    
    def get_active_sessions_count(self) -> int:
        """Get count of active sessions"""
        return len([s for s in self.sessions.values() if s.is_active])
    
    def get_user_session_info(self, username: str) -> Optional[Dict]:
        """Get session info for a specific user"""
        session_id = self.user_sessions.get(username)
        if session_id:
            return self.get_session_info(session_id)
        return None
    
    def is_user_logged_in(self, username: str) -> bool:
        """Check if a user has an active session"""
        session_id = self.user_sessions.get(username)
        if session_id:
            session = self.validate_session(session_id, update_activity=False)
            return session is not None
        return False

# Global session manager instance
session_manager = SessionManager(
    session_timeout_minutes=30,     # 30 minutes of inactivity
    max_session_lifetime_hours=24,  # 24 hours maximum session life
    cleanup_interval_minutes=5      # Cleanup every 5 minutes
)
