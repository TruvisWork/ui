import pandas as pd

df = pd.read_csv("input.csv")
df["period_start"] = pd.to_datetime(df["period_start"])
df["concurrent_slots"] = df["period_slot_ms"] / 1000

# Check for duplicates at same timestamp
print("Total rows:", len(df))
print("Unique timestamps:", df["period_start"].nunique())

# Find the peak timestamp and see what's there
peak_time = df.groupby("period_start")["concurrent_slots"].sum().idxmax()
peak_rows = df[df["period_start"] == peak_time]
print(f"\nRows at peak timestamp ({peak_time}):")
print(f"  Row count: {len(peak_rows)}")
print(f"  Sum of concurrent_slots: {peak_rows['concurrent_slots'].sum():.0f}")
print(f"  Max single row value: {peak_rows['concurrent_slots'].max():.0f}")

# Show column names
print("\nColumns in data:", df.columns.tolist())
