"""
================================================================================
BigQuery Slot Usage Analysis - Enhanced with Candlestick Charts
================================================================================

CONFIGURABLE TIMEFRAMES:
- 1s: Raw second-level data (default)
- 15min: 15-minute candles
- 4h: 4-hour candles
- 1d: Daily candles
- 7d: Weekly candles

CANDLESTICK INTERPRETATION (like stock market):
- Open: First slot value in the period
- High: Maximum slots in the period
- Low: Minimum slots in the period
- Close: Last slot value in the period

================================================================================
"""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Rectangle
import matplotlib.dates as mdates
from datetime import datetime
import argparse

# ==========================
# CONFIG
# ==========================
CSV_FILE = "input.csv"
OUTPUT_PREFIX = "slot_analysis"

# Timeframe options
TIMEFRAME_MAP = {
    "1s": "1s",
    "1min": "1min",
    "15min": "15min",
    "1h": "1h",
    "4h": "4h",
    "1d": "1D",
    "7d": "7D"
}

# Parse command line arguments
parser = argparse.ArgumentParser(description='BigQuery Slot Usage Analysis')
parser.add_argument('--timeframe', '-t', type=str, default='15min',
                    choices=list(TIMEFRAME_MAP.keys()),
                    help='Candlestick timeframe: 1s, 1min, 15min, 1h, 4h, 1d, 7d')
args = parser.parse_args()

CANDLE_TIMEFRAME = TIMEFRAME_MAP[args.timeframe]
print(f"Using candlestick timeframe: {args.timeframe}")

# ==========================
# LOAD DATA
# ==========================
print("="*70)
print("BIGQUERY SLOT USAGE ANALYSIS")
print("="*70)

df = pd.read_csv(CSV_FILE)

# Handle both column naming conventions
if "period_start" in df.columns and "period_slot_ms" in df.columns:
    df["time"] = pd.to_datetime(df["period_start"])
    df["concurrent_slots"] = df["period_slot_ms"] / 1000
elif "period_start" in df.columns and "concurrent_slots" in df.columns:
    df["time"] = pd.to_datetime(df["period_start"])
else:
    raise ValueError("Expected columns: period_start with period_slot_ms or concurrent_slots")

# Aggregate if multiple rows per timestamp
if df.groupby("time").size().max() > 1:
    concurrent = (
        df.groupby("time")["concurrent_slots"]
        .sum()
        .reset_index()
    )
    print("Note: Summing multiple rows per timestamp")
else:
    concurrent = df[["time", "concurrent_slots"]].copy()
    print("Note: Data already grouped")

concurrent = concurrent.set_index("time").sort_index()

# ==========================
# CALCULATE OHLC CANDLES
# ==========================
def create_ohlc(data, freq):
    """Create OHLC (Open, High, Low, Close) candlestick data"""
    ohlc = data["concurrent_slots"].resample(freq).agg({
        'open': 'first',
        'high': 'max',
        'low': 'min',
        'close': 'last',
        'avg': 'mean'
    })
    ohlc = ohlc.dropna()
    return ohlc

candles = create_ohlc(concurrent, CANDLE_TIMEFRAME)
print(f"Generated {len(candles)} candles at {args.timeframe} timeframe")

# ==========================
# METRICS
# ==========================
all_slots = concurrent["concurrent_slots"]
avg_slots = all_slots.mean()
p50 = np.percentile(all_slots, 50)
p90 = np.percentile(all_slots, 90)
p95 = np.percentile(all_slots, 95)
p98 = np.percentile(all_slots, 98)
p99 = np.percentile(all_slots, 99)
peak = all_slots.max()
min_slots = all_slots.min()

# ==========================
# TECHNICAL OUTPUT
# ==========================
print("\n" + "-"*70)
print("METRICS SUMMARY")
print("-"*70)
print(f"  Minimum:  {min_slots:>10.1f} slots")
print(f"  p50:      {p50:>10.1f} slots")
print(f"  Average:  {avg_slots:>10.1f} slots")
print(f"  p90:      {p90:>10.1f} slots")
print(f"  p95:      {p95:>10.1f} slots")
print(f"  p98:      {p98:>10.1f} slots")
print(f"  p99:      {p99:>10.1f} slots")
print(f"  Peak:     {peak:>10.1f} slots")

# ==========================
# GRAPH FUNCTIONS
# ==========================

def plot_candlestick(ax, ohlc_data, title):
    """Plot stock-market style candlestick chart"""
    width = 0.6  # candle width
    
    # Calculate width based on time difference
    if len(ohlc_data) > 1:
        time_diff = (ohlc_data.index[1] - ohlc_data.index[0]).total_seconds()
        width_td = pd.Timedelta(seconds=time_diff * 0.8)
    else:
        width_td = pd.Timedelta(hours=1)
    
    for idx, (time_idx, row) in enumerate(ohlc_data.iterrows()):
        open_val = row['open']
        close_val = row['close']
        high_val = row['high']
        low_val = row['low']
        
        # Color: green if close >= open, red if close < open
        if close_val >= open_val:
            color = '#26a69a'  # Green
            body_bottom = open_val
            body_height = close_val - open_val
        else:
            color = '#ef5350'  # Red
            body_bottom = close_val
            body_height = open_val - close_val
        
        # Draw wick (high-low line)
        ax.plot([time_idx, time_idx], [low_val, high_val], 
                color=color, linewidth=1, zorder=1)
        
        # Draw body (open-close rectangle)
        body_height = max(body_height, peak * 0.005)  # Min height for visibility
        rect = Rectangle(
            (mdates.date2num(time_idx) - mdates.date2num(time_idx + width_td)/2 + mdates.date2num(time_idx)/2, 
             body_bottom),
            width_td.total_seconds() / 86400,  # Convert to days for matplotlib
            body_height,
            facecolor=color,
            edgecolor=color,
            zorder=2
        )
        ax.add_patch(rect)
    
    # Add percentile lines
    ax.axhline(avg_slots, color='blue', linestyle='-', linewidth=2, label=f'Avg = {avg_slots:.0f}', alpha=0.8)
    ax.axhline(p50, color='green', linestyle='--', linewidth=1.5, label=f'p50 = {p50:.0f}', alpha=0.8)
    ax.axhline(p90, color='orange', linestyle='--', linewidth=1.5, label=f'p90 = {p90:.0f}', alpha=0.8)
    ax.axhline(p95, color='red', linestyle='--', linewidth=2, label=f'p95 = {p95:.0f}', alpha=0.8)
    ax.axhline(p99, color='darkred', linestyle=':', linewidth=1.5, label=f'p99 = {p99:.0f}', alpha=0.8)
    
    ax.set_title(title, fontsize=12, fontweight='bold')
    ax.set_xlabel("Time")
    ax.set_ylabel("Slots")
    ax.legend(loc='upper right', fontsize=8)
    ax.grid(alpha=0.3)
    ax.xaxis.set_major_formatter(mdates.DateFormatter('%m-%d %H:%M'))
    plt.setp(ax.xaxis.get_majorticklabels(), rotation=45, ha='right')


def plot_simplified_candlestick(ax, ohlc_data, title):
    """Simplified candlestick using bar chart with error bars for high/low"""
    times = range(len(ohlc_data))
    
    colors = ['#26a69a' if row['close'] >= row['open'] else '#ef5350' 
              for _, row in ohlc_data.iterrows()]
    
    # Plot as bars with the body (open to close)
    bottoms = [min(row['open'], row['close']) for _, row in ohlc_data.iterrows()]
    heights = [abs(row['close'] - row['open']) for _, row in ohlc_data.iterrows()]
    heights = [max(h, peak * 0.01) for h in heights]  # Min height
    
    ax.bar(times, heights, bottom=bottoms, color=colors, width=0.8, zorder=2)
    
    # Plot wicks
    for i, (_, row) in enumerate(ohlc_data.iterrows()):
        ax.plot([i, i], [row['low'], row['high']], color='black', linewidth=0.8, zorder=1)
    
    # Percentile lines
    ax.axhline(avg_slots, color='blue', linestyle='-', linewidth=2, label=f'Avg = {avg_slots:.0f}')
    ax.axhline(p50, color='green', linestyle='--', linewidth=1.5, label=f'p50 = {p50:.0f}')
    ax.axhline(p90, color='orange', linestyle='--', linewidth=1.5, label=f'p90 = {p90:.0f}')
    ax.axhline(p95, color='red', linestyle='--', linewidth=2, label=f'p95 = {p95:.0f}')
    ax.axhline(p99, color='darkred', linestyle=':', linewidth=1.5, label=f'p99 = {p99:.0f}')
    
    # X-axis labels (show subset)
    step = max(1, len(ohlc_data) // 10)
    ax.set_xticks(times[::step])
    ax.set_xticklabels([t.strftime('%m-%d %H:%M') for t in ohlc_data.index[::step]], 
                       rotation=45, ha='right', fontsize=8)
    
    ax.set_title(title, fontsize=12, fontweight='bold')
    ax.set_xlabel("Time")
    ax.set_ylabel("Slots")
    ax.legend(loc='upper right', fontsize=8)
    ax.grid(alpha=0.3)

ax4 = axes[1, 1]
