import pandas as pd
from datetime import datetime, timedelta

# Load your data
# df = pd.read_csv('your_data.csv')

# Sample data
data = {
    'start_time': ['2025-11-21 10:00:00', '2025-11-21 10:05:00', '2025-11-21 10:10:00'],
    'end_time': ['2025-11-21 10:30:00', '2025-11-21 10:15:00', '2025-11-21 10:40:00'],
    'slot_seconds': [3000, 1500, 5000],
    'query_type': ['procs', 'user_fired', 'sa_non_proc']
}
df = pd.DataFrame(data)

# Convert to datetime
df['start_time'] = pd.to_datetime(df['start_time'])
df['end_time'] = pd.to_datetime(df['end_time'])
df['duration_seconds'] = (df['end_time'] - df['start_time']).dt.total_seconds()
df['duration_seconds'] = df['duration_seconds'].replace(0, 1)
df['slots_used'] = df['slot_seconds'] / df['duration_seconds']

# REPLACE THIS WITH YOUR ACTUAL PEAK TIME FROM THE GRAPH
peak_time = pd.to_datetime('2025-12-01 20:07:00')

print("="*80)
print(f"ANALYZING PEAK USAGE AT: {peak_time}")
print("="*80)

# Find all queries running at peak time
queries_at_peak = df[
    (df['start_time'] <= peak_time) & 
    (df['end_time'] >= peak_time)
].copy()

print(f"\nTotal queries running at peak: {len(queries_at_peak)}")
print(f"Total concurrent slots at peak: {queries_at_peak['slots_used'].sum():.2f}")

# Sort by slot usage
queries_at_peak = queries_at_peak.sort_values('slots_used', ascending=False)

print("\n" + "="*80)
print("TOP 20 QUERIES BY SLOT USAGE AT PEAK TIME")
print("="*80)

for i, (idx, row) in enumerate(queries_at_peak.head(20).iterrows(), 1):
    duration_min = row['duration_seconds'] / 60
    print(f"\n{i}. Query Type: {row['query_type']}")
    print(f"   Slots Used: {row['slots_used']:.2f}")
    print(f"   Slot-Seconds: {row['slot_seconds']:,.0f}")
    print(f"   Duration: {duration_min:.2f} minutes")
    print(f"   Started: {row['start_time']}")
    print(f"   Ended: {row['end_time']}")
    if 'query' in df.columns:
        print(f"   Query: {row['query'][:100]}...")  # First 100 chars

# Statistics
print("\n" + "="*80)
print("STATISTICS OF QUERIES AT PEAK")
print("="*80)

print(f"\nSlot Usage Distribution:")
print(f"  Max single query: {queries_at_peak['slots_used'].max():.2f} slots")
print(f"  Mean: {queries_at_peak['slots_used'].mean():.2f} slots")
print(f"  Median: {queries_at_peak['slots_used'].median():.2f} slots")
print(f"  Top 10 queries account for: {queries_at_peak.head(10)['slots_used'].sum():.2f} slots")
print(f"  Top 10 as % of total: {(queries_at_peak.head(10)['slots_used'].sum() / queries_at_peak['slots_used'].sum() * 100):.1f}%")

print(f"\nBy Query Type at Peak:")
for qtype in ['procs', 'user_fired', 'sa_non_proc']:
    type_queries = queries_at_peak[queries_at_peak['query_type'] == qtype]
    if len(type_queries) > 0:
        print(f"  {qtype}: {len(type_queries)} queries, {type_queries['slots_used'].sum():.2f} slots")

# Find the spike window (queries that started around peak time)
window_start = peak_time - timedelta(minutes=30)
window_end = peak_time + timedelta(minutes=30)

queries_in_window = df[
    (df['start_time'] >= window_start) & 
    (df['start_time'] <= window_end)
].sort_values('slots_used', ascending=False)

print("\n" + "="*80)
print(f"QUERIES STARTED WITHIN 30 MINUTES OF PEAK ({window_start} to {window_end})")
print("="*80)
print(f"Total queries started in this window: {len(queries_in_window)}")

print("\nTop 10 heaviest queries started near peak:")
for i, (idx, row) in enumerate(queries_in_window.head(10).iterrows(), 1):
    print(f"{i}. Started: {row['start_time'].strftime('%H:%M:%S')} | "
          f"Type: {row['query_type']} | "
          f"Slots: {row['slots_used']:.2f} | "
          f"Duration: {row['duration_seconds']/60:.1f}min")

# Look for patterns - queries that might be scheduled together
print("\n" + "="*80)
print("PATTERN ANALYSIS")
print("="*80)

# Group by start time (rounded to 5 minutes) to find batches
df['start_rounded'] = df['start_time'].dt.floor('5min')
batch_analysis = df.groupby('start_rounded').agg({
    'slots_used': 'sum',
    'query_type': 'count'
}).rename(columns={'query_type': 'query_count'})
batch_analysis = batch_analysis.sort_values('slots_used', ascending=False)

print("\nTop 10 time periods with highest concurrent query starts (5-min windows):")
for time, row in batch_analysis.head(10).iterrows():
    print(f"  {time}: {row['query_count']} queries started, {row['slots_used']:.2f} total slots")

# Recommendations
print("\n" + "="*80)
print("RECOMMENDATIONS")
print("="*80)

top_query_slots = queries_at_peak.head(1)['slots_used'].values[0] if len(queries_at_peak) > 0 else 0
top_10_pct = (queries_at_peak.head(10)['slots_used'].sum() / queries_at_peak['slots_used'].sum() * 100) if len(queries_at_peak) > 0 else 0

print(f"""
1. OPTIMIZE TOP QUERIES
   - Your top single query uses {top_query_slots:.0f} slots
   - Top 10 queries account for {top_10_pct:.0f}% of peak usage
   - Focus optimization efforts on these queries

2. QUERY SCHEDULING
   - Check if multiple heavy queries start simultaneously around {peak_time.strftime('%H:%M')}
   - Consider staggering scheduled queries by 15-30 minutes

3. CAPACITY PLANNING
   - Current peak: {queries_at_peak['slots_used'].sum():.0f} slots
   - Normal usage appears to be ~5,000-10,000 slots
   - Consider: Burst capacity with autoscaling vs fixed reservation

4. IMMEDIATE ACTIONS
   - Review the queries listed above in your BigQuery console
   - Check for: missing WHERE clauses, unpartitioned table scans, large JOINs
   - Add LIMIT clauses for development/testing queries
   - Use clustering and partitioning on large tables
""")
