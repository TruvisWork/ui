import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
from datetime import datetime, timedelta
import numpy as np

# Load your data
# df = pd.read_csv('your_data.csv')

# Sample data structure - replace with your actual data
data = {
    'start_time': ['2025-11-21 10:00:00', '2025-11-21 10:05:00', '2025-11-21 10:10:00'],
    'end_time': ['2025-11-21 10:30:00', '2025-11-21 10:15:00', '2025-11-21 10:40:00'],
    'slot_seconds': [3000, 1500, 5000],
    'query_type': ['procs', 'user_fired', 'sa_non_proc']
}
df = pd.DataFrame(data)

print(f"Processing {len(df)} queries...")

# Convert to datetime
df['start_time'] = pd.to_datetime(df['start_time'])
df['end_time'] = pd.to_datetime(df['end_time'])

# Calculate slots per second for each query
df['duration_seconds'] = (df['end_time'] - df['start_time']).dt.total_seconds()
df['duration_seconds'] = df['duration_seconds'].replace(0, 1)  # Avoid division by zero
df['slots_used'] = df['slot_seconds'] / df['duration_seconds']

print(f"Slot usage range: {df['slots_used'].min():.2f} to {df['slots_used'].max():.2f}")

# Skip top chart, focus on concurrent usage which is more important
def calculate_concurrent_slots_optimized(df, bucket_size='15min'):
    """
    Calculate concurrent slot usage over time
    bucket_size: '15min', '30min', '1h', '4h' etc.
    """
    print(f"Using {bucket_size} time buckets...")
    
    # Get time range
    min_time = df['start_time'].min()
    max_time = df['end_time'].max()
    print(f"Time range: {min_time} to {max_time}")
    
    # Create time buckets
    time_range = pd.date_range(start=min_time.floor(bucket_size), 
                               end=max_time.ceil(bucket_size), 
                               freq=bucket_size)
    
    # Initialize results
    results = []
    
    # For each time bucket, calculate concurrent slot usage
    for i, bucket_start in enumerate(time_range[:-1]):
        bucket_end = time_range[i + 1]
        bucket_mid = bucket_start + (bucket_end - bucket_start) / 2
        
        # Find all queries running during this bucket
        running_queries = df[
            (df['start_time'] < bucket_end) & 
            (df['end_time'] > bucket_start)
        ]
        
        # Sum slots by type
        slots_by_type = {'procs': 0, 'user_fired': 0, 'sa_non_proc': 0}
        
        for _, query in running_queries.iterrows():
            # Calculate overlap duration
            overlap_start = max(query['start_time'], bucket_start)
            overlap_end = min(query['end_time'], bucket_end)
            overlap_duration = (overlap_end - overlap_start).total_seconds()
            
            # Add proportional slot usage
            if overlap_duration > 0:
                slots_by_type[query['query_type']] += query['slots_used']
        
        results.append({
            'time': bucket_mid,
            'procs': slots_by_type['procs'],
            'user_fired': slots_by_type['user_fired'],
            'sa_non_proc': slots_by_type['sa_non_proc'],
            'total': sum(slots_by_type.values()),
            'query_count': len(running_queries)
        })
    
    return pd.DataFrame(results)

print("\nCalculating concurrent slot usage with different granularities...")

# Calculate with multiple granularities
timeline_15min = calculate_concurrent_slots_optimized(df, '15min')
timeline_1h = calculate_concurrent_slots_optimized(df, '1h')
timeline_4h = calculate_concurrent_slots_optimized(df, '4h')

# Use 15min for main analysis, but show 4h overlay
timeline_df = timeline_15min

print(f"\nGenerated {len(timeline_df)} time points")
print(f"Concurrent slots range: {timeline_df['total'].min():.2f} to {timeline_df['total'].max():.2f}")
print(f"Non-zero periods: {(timeline_df['total'] > 0).sum()}")

# Create visualization - SINGLE CHART focused on concurrent usage
fig, ax = plt.subplots(1, 1, figsize=(20, 10))

colors = {'procs': '#e74c3c', 'user_fired': '#2ecc71', 'sa_non_proc': '#3498db'}

# Plot stacked area chart with 15-min granularity
ax.fill_between(timeline_df['time'], 0, timeline_df['procs'], 
                label='procs', color=colors['procs'], alpha=0.7, step='post')
ax.fill_between(timeline_df['time'], timeline_df['procs'], 
                timeline_df['procs'] + timeline_df['sa_non_proc'],
                label='sa_non_proc', color=colors['sa_non_proc'], alpha=0.7, step='post')
ax.fill_between(timeline_df['time'], 
                timeline_df['procs'] + timeline_df['sa_non_proc'],
                timeline_df['total'],
                label='user_fired', color=colors['user_fired'], alpha=0.7, step='post')

# Overlay 4-hour average as a thick line
ax.plot(timeline_4h['time'], timeline_4h['total'], 
        color='black', linewidth=3, label='4-Hour Average', linestyle='-', alpha=0.8)

# Find and mark peaks
# Top 5 peaks
top_peaks = timeline_df.nlargest(5, 'total')

for idx, peak in top_peaks.iterrows():
    ax.scatter([peak['time']], [peak['total']], color='red', s=150, zorder=5, 
               marker='o', edgecolors='black', linewidths=2)
    
    # Annotate only the highest peak
    if idx == timeline_df['total'].idxmax():
        ax.annotate(f"PEAK: {peak['total']:.1f} slots\n{peak['time'].strftime('%m/%d %H:%M')}\n({peak['query_count']} concurrent queries)",
                   xy=(peak['time'], peak['total']), 
                   xytext=(30, 30), textcoords='offset points',
                   bbox=dict(boxstyle='round,pad=0.8', fc='yellow', alpha=0.9),
                   arrowprops=dict(arrowstyle='->', connectionstyle='arc3,rad=0', lw=2),
                   fontsize=11, fontweight='bold')

# Add grid lines every 4 hours
ax.xaxis.set_major_locator(mdates.HourLocator(interval=4))
ax.xaxis.set_minor_locator(mdates.HourLocator(interval=1))
ax.xaxis.set_major_formatter(mdates.DateFormatter('%m/%d\n%H:%M'))

ax.set_xlabel('Time (MM/DD HH:MM)', fontsize=14, fontweight='bold')
ax.set_ylabel('Concurrent Slot Usage', fontsize=14, fontweight='bold')
ax.set_title(f'BigQuery Concurrent Slot Usage - 30 Days Analysis\n({len(df)} queries, 15-min resolution with 4-hour moving average)', 
            fontsize=16, fontweight='bold')
ax.legend(loc='upper left', fontsize=11)
ax.grid(True, alpha=0.3, which='major', linewidth=1.2)
ax.grid(True, alpha=0.15, which='minor', linewidth=0.5)

plt.setp(ax.xaxis.get_majorticklabels(), rotation=45, ha='right')
plt.tight_layout()
