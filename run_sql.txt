import os
from google.cloud.sql.connector import Connector
import pg8000
from .db_config import get_db_url_and_schema, get_instance_id
import argparse
import re
import logging

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def get_db_connection():
    try:
        instance_connection_name, database, iam_user, schema, input_dir, ip_type = get_db_url_and_schema()
        connector = Connector()
        conn = connector.connect(
            instance_connection_name,
            "pg8000",
            user=iam_user,
            db=database,
            enable_iam_auth=True,
            ip_type=ip_type,
        )
        return conn, connector
    except Exception as e:
        logger.error(f"Error: Unable to connect to the database. {e}")
        return None, None


def get_version_info(project_id="default_project", version_id_override=None):
    """Get version info for a project"""
    version_id = version_id_override
    
    if version_id:
        logger.info(f"Using provided version_id {version_id} for project {project_id}")
        return version_id
    else:
        logger.error("No version id provided")
        raise


def execute_sql_file(cursor, file_path, schema, instance_id, version_id, project_id):
    try:
        with open(file_path, 'r') as file:
            sql_query = file.read()
        sql_query = sql_query.format(
            schema=schema,
            instance=instance_id,
            version_id=version_id,
            project_id=project_id
        )
        logger.debug(f"Executing SQL from {file_path}")
        cursor.execute(sql_query)
        logger.info(f"Executed {file_path} for project {project_id}")
    except Exception as e:
        logger.error(f"Error executing {file_path}: {e}")
        raise


def get_ordered_sql_files(folder_path):
    try:
        if not os.path.exists(folder_path):
            return []
        sql_files = [f for f in os.listdir(folder_path) if f.endswith(".sql")]
        if not sql_files:
            return []
        numbered = sorted([f for f in sql_files if f[0].isdigit()], key=lambda f: int(re.match(r"(\d+)", f).group(1)))
        non_numbered = sorted([f for f in sql_files if not f[0].isdigit()])
        return numbered + non_numbered
    except Exception as e:
        logger.error(f"Error reading SQL dir {folder_path}: {e}")
        return []


def process_sql_files(folder_path, schema, instance_id, version_id, project_id):
    conn, connector = get_db_connection()
    if conn is None:
        return False
    cursor = conn.cursor()
    try:
        sql_files = get_ordered_sql_files(folder_path)
        if not sql_files:
            return True
        for filename in sql_files:
            file_path = os.path.join(folder_path, filename)
            execute_sql_file(cursor, file_path, schema, instance_id, version_id, project_id)
        conn.commit()
        logger.info(f"Executed {len(sql_files)} files for project {project_id}")
        return True
    except Exception as e:
        logger.error(f"Error processing SQL for project {project_id}: {e}")
        conn.rollback()
        return False
    finally:
        cursor.close()
        conn.close()
        if connector: connector.close()


def get_sql_directories(base_path):
    return {
        'input_sql_dir': os.path.join(base_path, 'unified_data_load_scripts'),
        'output_sql_dir1': os.path.join(base_path, 'recommendation_data_load_scripts'),
        'output_sql_dir2': os.path.join(base_path, 'recommendation_data_load_scripts_2'),
        'output_sql_dir3': os.path.join(base_path, 'recommendation_summary_dl_scripts')
    }


def main():
    parser = argparse.ArgumentParser(description="Run SQL files with version tracking.")
    parser.add_argument("mode", choices=["input", "output"], help="Mode: input or output")
    parser.add_argument("--config-path", default=None, help="Config file path")
    parser.add_argument("--project-id", default="default_project", help="Single project ID (for input mode)")
    parser.add_argument("--project-ids", default=None, help="Comma-separated project IDs (for output mode)")
    parser.add_argument("--version-id", default=None, type=int, help="Version ID")

    args = parser.parse_args()
    if args.config_path:
        os.environ["CONFIG_PATH"] = args.config_path

    try:
        _, _, _, schema, _, _ = get_db_url_and_schema()
        instance_id = get_instance_id()
    except Exception as e:
        logger.error(f"Error getting DB config: {e}")
        return 1

    # Get version info
    primary_project = args.project_id if args.mode == "input" else (args.project_ids.split(",")[0] if args.project_ids else "default_project")
    version_id, refresh_date = get_version_info(primary_project, args.version_id)

    if not version_id:
        logger.error(f"Could not determine version ID. Aborting.")
        return 1

    base_path = os.path.dirname(os.path.abspath(__file__))
    sql_dirs = get_sql_directories(base_path)
    success = True

    if args.mode == "input":
        # Input mode - single project ID
        input_dir = sql_dirs['input_sql_dir']
        if not os.path.exists(input_dir):
            logger.error(f"Input SQL directory does not exist: {input_dir}")
            return 1
        if not process_sql_files(input_dir, schema, instance_id, version_id, refresh_date, args.project_id):
            success = False

    elif args.mode == "output":
        # Output mode - all projects combined (use project_ids parameter)
        project_ids_str = args.project_ids if args.project_ids else "default_project"
        for sql_dir in [sql_dirs['output_sql_dir1'], sql_dirs['output_sql_dir2'], sql_dirs['output_sql_dir3']]:
            if os.path.exists(sql_dir):
                if not process_sql_files(sql_dir, schema, instance_id, version_id, refresh_date, project_ids_str):
                    success = False
                    break

    return 0 if success else 1

if __name__ == "__main__":
    exit(main())
