def create_access_token(data: dict, expires_delta: Optional[timedelta] = None):
    to_encode = data.copy()
    expire = datetime.utcnow() + (expires_delta or timedelta(minutes=15))
    to_encode.update({"exp": expire})
    logger.info("CREATE_TOKEN - Data being encoded: %s", to_encode)
    token = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    logger.info("CREATE_TOKEN - Token created successfully")
    return token

def verify_token(request: Request, credentials: HTTPAuthorizationCredentials = Depends(security)):
    """
    Verify token from either:
    1. Cookie (preferred)
    2. Authorization header (Bearer token)
    """
    token = None
    
    try:
        logger.info("VERIFY_TOKEN_START - Checking for token")
        
        # First try to get token from cookie
        token = request.cookies.get("access_token")
        logger.info("VERIFY_TOKEN_COOKIE - Token found in cookie: %s", bool(token))
        
        # If no cookie, try Authorization header
        if not token and credentials:
            token = credentials.credentials
            logger.info("VERIFY_TOKEN_HEADER - Token found in header: %s", bool(token))
        
        if not token:
            logger.warning("TOKEN_VERIFICATION_FAILED - No token provided")
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="No authentication token provided",
                headers={"WWW-Authenticate": "Bearer"},
            )
        
        logger.info("VERIFY_TOKEN_DECODE - Attempting to decode token")
        # Decode and verify token
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        logger.info("VERIFY_TOKEN_PAYLOAD - Payload decoded successfully: %s", payload)
        
        username: str = payload.get("sub")
        markets: List[str] = payload.get("markets", [])
        uid: str = payload.get("uid")
        
        if not username:
            logger.warning("TOKEN_VERIFICATION_FAILED - No username in token")
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: no username",
                headers={"WWW-Authenticate": "Bearer"},
            )
        
        logger.debug("TOKEN_VERIFICATION_SUCCESS - User: %s, Markets: %s, UID: %s", username, markets, uid)
        return {"username": username, "markets": markets, "uid": uid}
        
    except jwt.ExpiredSignatureError:
        logger.warning("TOKEN_VERIFICATION_FAILED - Token expired")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Token has expired",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except jwt.PyJWTError as e:
        logger.warning("TOKEN_VERIFICATION_FAILED - JWT error: %s", str(e))
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except Exception as e:
        logger.exception("TOKEN_VERIFICATION_ERROR - Unexpected error: %s", str(e))
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Internal server error during token verification"
        )

def find_user_by_identifier(identifier: str) -> Optional[dict]:
    """Find user by username or email from the mock LDAP users"""
    for username, user_data in ldap_auth.MOCK_USERS.items():
        user_attributes = user_data.get("attributes", {})
        email = user_attributes.get("mail", [None])[0]
        
        if (identifier.lower() == username.lower() or 
            (email and identifier.lower() == email.lower())):
            return {
                "username": username,
                "email": email,
                "name": user_attributes.get("displayName", [username])[0],
                "markets": extract_markets_from_groups(user_attributes.get("memberOf", []))
            }
    return None

def extract_markets_from_groups(member_of):
    """Extract market codes from LDAP group membership"""
    markets = []
    market_prefix = "cn="
    market_suffix = "_market,ou=markets,"
    
    for group in member_of:
        if market_suffix in group:
            try:
                market = group.split(market_prefix)[1].split("_market")[0].upper()
                if market and market not in markets:
                    markets.append(market)
            except (IndexError, AttributeError):
                logger.warning("Failed to extract market from group: %s", group)
                continue
    
    logger.info("EXTRACT_MARKETS - Groups: %s, Markets: %s", member_of, markets)
    return markets

def authenticate(username: str, password: str) -> Optional[Dict]:
    """Authenticate user against mock LDAP"""
    logger.info("AUTHENTICATE - Attempting to authenticate user: %s", username)
    
    user = ldap_auth.MOCK_USERS.get(username)
    if user and user["password"] == password:
        logger.info("AUTHENTICATE_SUCCESS - User %s authenticated", username)
        return {
            "username": username,
            "dn": user["attributes"]["dn"],
            "email": user["attributes"]["mail"][0],
            "displayName": user["attributes"]["displayName"][0],
            "department": user["attributes"]["department"][0],
            "title": user["attributes"]["title"][0],
            "groups": user["attributes"]["memberOf"],
            "attributes": user["attributes"],
            "authenticated_at": datetime.utcnow().isoformat()
        }
    
    logger.warning("AUTHENTICATE_FAILED - Invalid credentials for user: %s", username)
    return None
