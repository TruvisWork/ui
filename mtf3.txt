# ------------------------------------------------------------------------------
# GCS Resources
# ------------------------------------------------------------------------------
resource "google_storage_bucket_object" "startup_script" {
  count  = var.build_vm ? 1 : 0
  name   = var.startup_script_path
  bucket = local.startup_script_bucket
  source = var.startup_script_local_path

  detect_md5hash = "different"

  lifecycle {
    create_before_destroy = true
  }
}

# ------------------------------------------------------------------------------
# VM Pre-Creation Deletion
# This ensures the VM is always deleted before creation (shell script behavior)
# ------------------------------------------------------------------------------
resource "null_resource" "delete_instance_before_create" {
  count = var.build_vm ? 1 : 0

  triggers = {
    # This ensures the VM is recreated every time
    always_run = timestamp()
  }

  provisioner "local-exec" {
    command = "gcloud compute instances delete ${local.vm_instance_name} --project=${var.my_project} --zone=${local.zone} --quiet --impersonate-service-account=${local.build_account_email} || true"
  }
}

# ------------------------------------------------------------------------------
# Resource Policy
# ------------------------------------------------------------------------------
resource "null_resource" "check_policy_exists" {
  count = var.build_vm ? 1 : 0

  provisioner "local-exec" {
    command = <<-EOT
      if gcloud compute resource-policies describe ${local.policy_instance_name} --region=${var.region} --project=${var.my_project} --impersonate-service-account=${local.build_account_email} > /dev/null 2>&1; then
        echo "Resource policy ${local.policy_instance_name} already exists"
      else
        echo "Creating resource policy ${local.policy_instance_name}"
        gcloud compute resource-policies create instance-schedule ${local.policy_instance_name} \
          --region=${var.region} \
          --vm-stop-schedule="${var.vm_stop_schedule}" \
          --timezone=${local.timezone} \
          --description="Office Hours for ${var.inst_postfix}" \
          --project=${var.my_project} \
          --impersonate-service-account=${local.build_account_email}
      fi
    EOT
  }
}

# ------------------------------------------------------------------------------
# GCE Proxy VM Instance
# ------------------------------------------------------------------------------
resource "google_compute_instance" "proxy_vm" {
  count = var.build_vm ? 1 : 0

  depends_on = [null_resource.delete_instance_before_create]

  project      = var.my_project
  zone         = local.zone
  name         = local.vm_instance_name
  machine_type = var.machine_type
  tags         = var.network_tags

  boot_disk {
    initialize_params {
      image = "projects/${var.image_project}/global/images/family/${var.image_family_stage2}"
      size  = var.disk_size_gb
      type  = var.disk_type
    }
    device_name       = local.vm_instance_name
    kms_key_self_link = local.kms_compute_key
  }

  network_interface {
    subnetwork = local.host_subnet_uri
    network_ip = data.google_compute_address.static_ip.address
  }

  network_interface {
    subnetwork = local.def_subnet_uri
  }

  service_account {
    email  = local.vm_sa_email
    scopes = var.vm_scopes
  }

  metadata = {
    startup-script-url     = "gs://${local.startup_script_bucket}/${google_storage_bucket_object.startup_script[0].name}"
    region                 = var.region
    zonePostfix            = var.zone_postfix
    project                = var.my_project
    yumpackages            = var.yum_packages
    env                    = local.env
    nexus3                 = "nexus3"
    gbmtNexus              = "gbmt-nexus"
    nexus302               = "nexus302"
    bucketUriStartup       = "gs://${local.startup_script_bucket}"
    cloudsql_instance_name = local.sql_instance_name
    cloudsql_private_ip    = var.build_sql ? google_sql_database_instance.main_db[0].private_ip_address : ""
    isGmiTest              = "true"
  }

  allow_stopping_for_update = true

  lifecycle {
    ignore_changes = [scheduling]
  }
}

# ------------------------------------------------------------------------------

resource "google_compute_instance_iam_member" "vm_stopper" {
  count         = var.build_vm ? 1 : 0
  project       = var.my_project
  zone          = local.zone
  instance_name = google_compute_instance.proxy_vm[0].name
  role          = "organizations/103829057055/roles/cr.instancestopStart"
  member        = "serviceAccount:service-${data.google_project.project.number}@compute-system.iam.gserviceaccount.com"
}

# ------------------------------------------------------------------------------
# Attach Resource Policy
# ------------------------------------------------------------------------------
resource "null_resource" "add_vm_schedule" {
  count = var.build_vm ? 1 : 0
  depends_on = [
    google_compute_instance.proxy_vm,
    google_compute_instance_iam_member.vm_stopper,
    null_resource.check_policy_exists
  ]

  provisioner "local-exec" {
    command = "gcloud compute instances add-resource-policies ${google_compute_instance.proxy_vm[0].name} --resource-policies=${local.policy_instance_name} --zone=${local.zone} --project=${var.my_project} --impersonate-service-account=${local.build_account_email} || true"
  }
}

# ------------------------------------------------------------------------------
# Cloud SQL Instance - Native Terraform Resource
# ------------------------------------------------------------------------------
resource "google_sql_database_instance" "main_db" {
  count            = var.build_sql ? 1 : 0
  name             = local.sql_instance_name
  project          = var.my_project
  database_version = "POSTGRES_13"
  region           = var.region

  settings {
    tier              = "db-custom-2-8192"
    availability_type = "REGIONAL"
    disk_type         = "PD_SSD"
    disk_size         = 10

    backup_configuration {
      enabled    = true
      start_time = "04:00"
      binary_log_enabled = true
    }

    maintenance_window {
      day  = 7  # Sunday
      hour = 4
    }

    ip_configuration {
      ipv4_enabled    = false
      private_network = "projects/${var.my_project}/global/networks/${var.default_network}"
    }

    database_flags {
      name  = "cloudsql.iam_authentication"
      value = "on"
    }
  }

  encryption_key_name = local.kms_sql_key

  lifecycle {
    prevent_destroy = true
    ignore_changes = [
      settings[0].disk_size
    ]
  }
}
