import { describe, it, expect, vi, beforeEach } from 'vitest';
import { render, screen, waitFor } from '@testing-library/react';

// Mock the components
vi.mock('./components/Insight', () => ({
    default: () => <div>Insight Component</div>
}));

vi.mock('./components/Optimize', () => ({
    default: () => <div>Optimize Component</div>
}));

vi.mock('./components/Recommendation', () => ({
    default: () => <div>Recommendation Component</div>
}));

vi.mock('./components/DataCatalogEditor', () => ({
    default: () => <div>DataCatalogEditor Component</div>
}));

vi.mock('./components/Auth', () => ({
    default: () => <div>Auth Component</div>
}));

vi.mock('./components/QueryDetailsPage', () => ({
    default: () => <div>QueryDetailsPage Component</div>
}));

describe('App Component', () => {
    beforeEach(() => {
        sessionStorage.clear();
        vi.clearAllMocks();
    });

    it('should handle sessionStorage operations', () => {
        sessionStorage.setItem('selectedMarket', 'US');
        expect(sessionStorage.getItem('selectedMarket')).toBe('US');

        sessionStorage.removeItem('selectedMarket');
        expect(sessionStorage.getItem('selectedMarket')).toBeNull();
    });

    it('should store and retrieve user information', () => {
        sessionStorage.setItem('userName', 'testuser');
        sessionStorage.setItem('projectId', 'test-project-123');
        sessionStorage.setItem('selectedMarket', 'UK');

        expect(sessionStorage.getItem('userName')).toBe('testuser');
        expect(sessionStorage.getItem('projectId')).toBe('test-project-123');
        expect(sessionStorage.getItem('selectedMarket')).toBe('UK');
    });

    it('should clear all session data', () => {
        sessionStorage.setItem('allowedMarkets', 'US,UK');
        sessionStorage.setItem('selectedMarket', 'US');
        sessionStorage.setItem('username', 'john');

        sessionStorage.clear();

        expect(sessionStorage.getItem('allowedMarkets')).toBeNull();
        expect(sessionStorage.getItem('selectedMarket')).toBeNull();
        expect(sessionStorage.getItem('username')).toBeNull();
    });

    it('should validate protected route logic', () => {
        const selectedMarket = sessionStorage.getItem('selectedMarket');
        expect(selectedMarket).toBeNull();

        sessionStorage.setItem('selectedMarket', 'US');
        const marketAfterSet = sessionStorage.getItem('selectedMarket');
        expect(marketAfterSet).toBe('US');
    });

    it('should handle market change request flag', () => {
        sessionStorage.setItem('changeMarketRequest', 'true');
        expect(sessionStorage.getItem('changeMarketRequest')).toBe('true');

        sessionStorage.removeItem('changeMarketRequest');
        expect(sessionStorage.getItem('changeMarketRequest')).toBeNull();
    });
});
