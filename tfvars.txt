        String sql = "SELECT a.store_id, b.store_cost FROM foodmart.sales_fact_1997 a " +
                "JOIN foodmart.sales_fact_1998 b ON a.product_id = b.product_id " +
                "JOIN foodmart.sales_fact_dec_1998 c ON a.product_id = c.product_id "  +
                "JOIN foodmart.sales_fact_dec_1999 d ON b.product_id = d.product_id";
        SqlNode sqlNode = parseSql(sql);

        });

                .filter(rel -> rel.getRelationshipType() == RelationshipType.JOINS)
                .count();

     /*   assertNotNull(result.getContext().getJoinClauses(), "Join clauses should not be null");

    @Test
    void testJoinColumnsSingleEquality() throws Exception {
        String sql = "SELECT a.store_id, b.store_cost FROM foodmart.sales_fact_1997 a " +
                "JOIN foodmart.sales_fact_1998 b ON a.product_id = b.product_id";
        SqlNode sqlNode = parseSql(sql);

        SqlNodeVisitor.Result result = sqlNode.accept(visitor);
        assertNotNull(result);
/*        assertNotNull(result.getContext().getJoinClauses());
        assertEquals(1, result.getContext().getJoinClauses().size(), "Should have 1 join clause");*/

        var joinColumns = result.getColumnInfos().stream()
                .filter(ci -> ci.getClause() == ClauseType.JOIN)
                .toList();
        assertEquals(2, joinColumns.size(), "Single equality join should capture two column entries (left and right)");

        boolean has1997 = joinColumns.stream().anyMatch(c -> "sales_fact_1997".equals(c.getTableName()) && "product_id".equals(c.getColumnName()));
        boolean has1998 = joinColumns.stream().anyMatch(c -> "sales_fact_1998".equals(c.getTableName()) && "product_id".equals(c.getColumnName()));
        assertTrue(has1997, "Join columns should include sales_fact_1997.product_id");
        assertTrue(has1998, "Join columns should include sales_fact_1998.product_id");
    }

    @Test
    void testJoinColumnsMultiColumnAnd() throws Exception {
        String sql = "SELECT * FROM foodmart.sales_fact_1997 a " +
                "JOIN foodmart.sales_fact_1998 b ON a.product_id = b.product_id AND a.store_id = b.store_id";
        SqlNode sqlNode = parseSql(sql);

        SqlNodeVisitor.Result result = sqlNode.accept(visitor);
        assertNotNull(result);
  /*      assertNotNull(result.getContext().getJoinClauses());
        assertEquals(1, result.getContext().getJoinClauses().size(), "Should have 1 join clause");*/

        var joinColumns = result.getColumnInfos().stream()
                .filter(ci -> ci.getClause() == ClauseType.JOIN)
                .toList();
        assertEquals(4, joinColumns.size(), "ANDed two-column join should report 4 entries (2 per side)");

        String[] tables = {"sales_fact_1997", "sales_fact_1998"};
        String[] cols = {"product_id", "store_id"};
        for (String t : tables) {
            for (String c : cols) {
                boolean present = joinColumns.stream().anyMatch(jc -> t.equals(jc.getTableName()) && c.equals(jc.getColumnName()));
                assertTrue(present, "Expected join column present: " + t + "." + c);
            }
        }

        long joinRelationships = result.getEntityRelationships().stream()
                .filter(rel -> rel.getRelationshipType() == RelationshipType.JOINS)
                .count();
        assertEquals(1, joinRelationships, "Should only create one join relationship for two-table join");
    }

    @Test
    void testTableOccurrenceTracking() throws Exception {
        // Same table used in main query and subquery with different filters
        // Columns from main query should have occurrence 1, columns from subquery should have occurrence 2
        String sql = "SELECT product_id, store_sales " +
                "FROM foodmart.sales_fact_1997 " +
                "WHERE store_sales > 100 " +
                "AND product_id IN (SELECT product_id FROM foodmart.sales_fact_1997 WHERE store_cost < 50)";
        SqlNode sqlNode = parseSql(sql);

        SqlNode enrichedSqlNode = sqlNode.accept(enricher);
        SqlNodeVisitor.Result result = enrichedSqlNode.accept(visitor);

        assertNotNull(result);
        assertEquals(StatementType.SELECT, result.getStatementType());

        // Verify that the same table appears multiple times in sourceTables
        assertTrue(result.getSourceTables().size() >= 2,
                "Same table should appear at least twice in sourceTables (main query + subquery)");

        // Get all column infos
        List<ColumnInfo> allColumns = result.getColumnInfos();
        assertFalse(allColumns.isEmpty(), "Should have column infos");

        // Verify WHERE clause columns from main query and subquery have different occurrences
        List<ColumnInfo> whereColumns = allColumns.stream()
                .filter(ci -> ci.getClause() == ClauseType.WHERE)
                .filter(ci -> "sales_fact_1997".equalsIgnoreCase(ci.getTableName()))
                .toList();

        assertFalse(whereColumns.isEmpty(), "Should have WHERE clause columns from sales_fact_1997");

        // At least one column should have occurrence > 1
        // Main query columns should have occurrence 1, subquery columns should have occurrence 2
        boolean hasOccurrenceGreaterThanOne = whereColumns.stream()
                .anyMatch(ci -> ci.getOccurrence() != null && ci.getOccurrence() > 1);

        assertTrue(hasOccurrenceGreaterThanOne,
                "At least one column should have occurrence > 1 when same table appears in main query and subquery. " +
                "Found WHERE clause occurrences: " + whereColumns.stream()
                        .map(ci -> String.format("%s.%s=%d", ci.getTableName(), ci.getColumnName(), ci.getOccurrence()))
                        .toList());

        // Verify we have columns with occurrence 1 (from main query)
        boolean hasOccurrenceOne = whereColumns.stream()
                .anyMatch(ci -> ci.getOccurrence() != null && ci.getOccurrence() == 1);

        assertTrue(hasOccurrenceOne,
                "Should have at least one column with occurrence 1 (from main query). " +
                "Found occurrences: " + whereColumns.stream()
                        .map(ci -> String.format("%s.%s=%d", ci.getTableName(), ci.getColumnName(), ci.getOccurrence()))
                        .toList());

        // Debug output: print all column infos with their occurrences
        System.out.println("=== Column Occurrence Debug ===");
        System.out.println("Total sourceTables: " + result.getSourceTables().size());
        allColumns.forEach(ci -> System.out.println(
                String.format("Clause: %s, Table: %s, Column: %s, Occurrence: %s",
                        ci.getClause(), ci.getTableName(), ci.getColumnName(), ci.getOccurrence())));
    }

    private String getTableNameFromIdentifier(SqlIdentifier id) {
        if (id.names != null && !id.names.isEmpty()) {
            return id.names.get(id.names.size() - 1);
        }
        return null;
    }

    private SqlNode parseSql(String sql) throws Exception {
        BigQuerySqlParser parser = new BigQuerySqlParser();
        return parser.parse(sql);
    }
