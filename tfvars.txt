import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==========================
# CONFIG
# ==========================
CSV_FILE = "input.csv"      # works for both datasets
TIME_RESOLUTION = "1min"    # can change to 5min if needed

# ==========================
# LOAD DATA
# ==========================
df = pd.read_csv(CSV_FILE)

# ==========================
# DETECT DATASET TYPE
# ==========================
if {"period_start", "period_slot_ms"}.issubset(df.columns):
    DATASET_TYPE = "TIMELINE"
elif {"start_time", "end_time", "slot_seconds"}.issubset(df.columns):
    DATASET_TYPE = "PROJECT"
else:
    raise ValueError("Unsupported dataset format")

print(f"Detected dataset type: {DATASET_TYPE}")

# ==========================
# BUILD CONCURRENT SLOTS
# ==========================
if DATASET_TYPE == "TIMELINE":
    # JOBS_TIMELINE path (TRUE concurrency)
    df["period_start"] = pd.to_datetime(df["period_start"])
    df["concurrent_slots"] = df["period_slot_ms"] / 1000

    concurrent = (
        df.groupby("period_start")["concurrent_slots"]
          .sum()
          .resample(TIME_RESOLUTION)
          .mean()
          .reset_index()
    )

else:
    # JOBS_BY_PROJECT path (estimated concurrency)
    df["start_time"] = pd.to_datetime(df["start_time"])
    df["end_time"] = pd.to_datetime(df["end_time"])

    df["duration_sec"] = (df["end_time"] - df["start_time"]).dt.total_seconds()
    df["avg_slots"] = df["slot_seconds"] / df["duration_sec"]

    time_index = pd.date_range(
        start=df["start_time"].min(),
        end=df["end_time"].max(),
        freq=TIME_RESOLUTION
    )

    rows = []
    for t in time_index:
        active = df[(df["start_time"] <= t) & (df["end_time"] > t)]
        rows.append({"time": t, "concurrent_slots": active["avg_slots"].sum()})

    concurrent = pd.DataFrame(rows)

# ==========================
# METRICS
# ==========================
p50 = np.percentile(concurrent["concurrent_slots"], 50)
p90 = np.percentile(concurrent["concurrent_slots"], 90)
p95 = np.percentile(concurrent["concurrent_slots"], 95)
peak = concurrent["concurrent_slots"].max()

# ==========================
# GRAPH 1: CONCURRENCY TIMELINE
# ==========================
plt.figure(figsize=(14,6))
plt.plot(concurrent.iloc[:,0], concurrent["concurrent_slots"])
plt.axhline(p95, linestyle="--", label="p95 (Reserve Target)")
plt.axhline(p90, linestyle=":", label="p90")
plt.title("Concurrent Slot Usage Over Time")
plt.xlabel("Time")
plt.ylabel("Concurrent Slots")
plt.legend()
plt.tight_layout()
plt.show()

# ==========================
# GRAPH 2: DISTRIBUTION
# ==========================
plt.figure(figsize=(10,5))
plt.hist(concurrent["concurrent_slots"], bins=50)
plt.axvline(p50, linestyle=":", label="p50")
plt.axvline(p90, linestyle="--", label="p90")
plt.axvline(p95, linestyle="--", label="p95")
plt.axvline(peak, linestyle="-.", label="Peak")
plt.title("Distribution of Concurrent Slot Usage")
plt.xlabel("Concurrent Slots")
plt.ylabel("Frequency")
plt.legend()
plt.tight_layout()
plt.show()

# ==========================
# GRAPH 3: 24-HOUR AVERAGE
# ==========================
concurrent["hour"] = concurrent.iloc[:,0].dt.hour
hourly_avg = concurrent.groupby("hour")["concurrent_slots"].mean()

plt.figure(figsize=(10,5))
plt.plot(hourly_avg.index, hourly_avg.values)
plt.title("Average Slot Usage by Hour of Day")
plt.xlabel("Hour of Day")
plt.ylabel("Average Slots")
plt.xticks(range(0,24))
plt.tight_layout()
plt.show()

# ==========================
# SUMMARY
# ==========================
print("\n===== SLOT ANALYSIS SUMMARY =====")
print(f"p50  : {p50:.1f}")
print(f"p90  : {p90:.1f}")
print(f"p95  : {p95:.1f}  <-- Recommended Reserved Slots")
print(f"Peak : {peak:.1f}")
