# ------------------------------------------------------------------------------
# GCS Resources
# ------------------------------------------------------------------------------
resource "google_storage_bucket_object" "startup_script" {
  count  = local.build_vm ? 1 : 0
  name   = local.startup_script_path
  bucket = local.startup_script_bucket
  source = local.startup_script_local_path

    # Only create if doesn't exist, or update if content changed
    create_before_destroy = true
  }
}

# ------------------------------------------------------------------------------
# Pre-Creation Deletion Steps
# This section ensures the VM is deleted before creation, mirroring the script.
# ------------------------------------------------------------------------------
resource "null_resource" "delete_instance_before_create" {
  count = local.build_vm ? 1 : 0

  triggers = {
    # This ensures the VM is recreated every time
    always_run = timestamp()
  }

  provisioner "local-exec" {
    command = "gcloud compute instances delete ${local.vm_instance_name} --project=${local.my_project} --zone=${local.zone} --quiet --impersonate-service-account=${local.build_account_email} || true"
  }
}

# --- VM Resource Policy for Scheduling ---
# Only create if it doesn't exist
resource "google_compute_resource_policy" "vm_schedule" {
  count       = local.build_vm && length(data.google_compute_resource_policy.existing_policy) == 0 ? 1 : 0
  project     = local.my_project
  name        = local.policy_instance_name
  region      = local.region
  description = "Office Hours for ${local.inst_postfix}"

  instance_schedule_policy {
    time_zone = local.timezone
    vm_stop_schedule {
      schedule = local.vm_stop_schedule
    }
  }

  lifecycle {
    # Prevent destroy if resource already exists
    prevent_destroy = false
  }
}

# --- GCE Proxy VM Instance (from create-vm.sh) ---
resource "google_compute_instance" "proxy_vm" {
  count = local.build_vm ? 1 : 0

  # This ensures the deletion command runs before this resource is created.
  depends_on = [null_resource.delete_instance_before_create]

  project      = local.my_project
  zone         = local.zone
  name         = local.vm_instance_name
  machine_type = local.machine_type
  tags         = local.network_tags

  boot_disk {
    initialize_params {
      image = "projects/${local.image_project}/global/images/family/${local.image_family_stage2}"
      size  = local.disk_size_gb
      type  = local.disk_type
    }
    device_name      = local.vm_instance_name
    kms_key_self_link = local.kms_compute_key
  }

  network_interface {
    subnetwork = local.host_subnet_uri
    network_ip = data.google_compute_address.static_ip.address
  }

  network_interface {
    subnetwork = local.def_subnet_uri
  }

  service_account {
    email  = local.vm_sa_email
    scopes = local.vm_scopes
  }

  metadata = {
    startup-script-url     = "gs://${local.startup_script_bucket}/${google_storage_bucket_object.startup_script[0].name}"
    region                 = local.region
    zonePostfix            = local.zone_postfix
    project                = local.my_project
    yumpackages            = local.yum_packages
    env                    = local.env
    nexus3                 = "nexus3"
    gbmtNexus              = "gbmt-nexus"
    nexus302               = "nexus302"
    bucketUriStartup       = "gs://${local.startup_script_bucket}"
    cloudsql_instance_name = local.sql_instance_name
    cloudsql_private_ip    = local.build_sql ? (length(data.google_sql_database_instance.existing_db) > 0 ? data.google_sql_database_instance.existing_db[0].private_ip_address : google_sql_database_instance.main_db[0].private_ip_address) : ""
    isGmiTest              = "true"
  }

  allow_stopping_for_update = true

  lifecycle {
    ignore_changes = [scheduling]
  }
}

# --- IAM Binding for the Proxy VM ---
resource "google_compute_instance_iam_member" "vm_stopper" {
  count         = local.build_vm ? 1 : 0
  project       = local.my_project
  zone          = local.zone
  instance_name = google_compute_instance.proxy_vm[0].name
  role          = "organizations/103829057055/roles/cr.instancestopStart"
  member        = "serviceAccount:service-${data.google_project.project.number}@compute-system.iam.gserviceaccount.com"
}

# --- Attach the Resource Policy using local-exec ---
resource "null_resource" "add_vm_schedule" {
  count = local.build_vm ? 1 : 0
  depends_on = [
    google_compute_instance.proxy_vm,
    google_compute_instance_iam_member.vm_stopper
  ]
  
  provisioner "local-exec" {
    # Use the existing policy if it exists, otherwise use the newly created one
    command = <<-EOT
      POLICY_NAME=${length(data.google_compute_resource_policy.existing_policy) > 0 ? data.google_compute_resource_policy.existing_policy[0].name : (length(google_compute_resource_policy.vm_schedule) > 0 ? google_compute_resource_policy.vm_schedule[0].name : local.policy_instance_name)}
      gcloud compute instances add-resource-policies ${google_compute_instance.proxy_vm[0].name} \
        --resource-policies=$POLICY_NAME \
        --zone=${local.zone} \
        --project=${local.my_project} \
        --impersonate-service-account=${local.build_account_email} || true
    EOT
  }
}

  deletion_protection = false

  settings {
    tier              = "db-custom-2-8192"
    activation_policy = "ALWAYS"
    availability_type = "REGIONAL"
    disk_autoresize   = true
    disk_type         = "PD_SSD"

    ip_configuration {
      ipv4_enabled    = false
      private_network = "projects/${data.google_project.project.name}/global/networks/${local.default_network}"
    }

    backup_configuration {
      enabled = true
    }

    maintenance_window {
      day  = 1 # Sunday
      hour = 4
    }

    insights_config {
      query_insights_enabled = true
    }
    
    database_flags {
      name  = "cloudsql.iam_authentication"
      value = "on"
    }
  }

  lifecycle {
    prevent_destroy = false
  }
}

# --- Google Cloud SQL Database ---
resource "google_sql_database" "database" {
  count    = local.build_sql ? 1 : 0
  instance = length(data.google_sql_database_instance.existing_db) > 0 ? data.google_sql_database_instance.existing_db[0].name : google_sql_database_instance.main_db[0].name
  name     = local.db_name

  lifecycle {
    # Don't fail if database already exists
    create_before_destroy = false
  }
}

# --- Google Cloud SQL User ---
resource "google_sql_user" "postgres_admin" {
  count    = local.build_sql ? 1 : 0
  instance = length(data.google_sql_database_instance.existing_db) > 0 ? data.google_sql_database_instance.existing_db[0].name : google_sql_database_instance.main_db[0].name
  name     = "postgres"
  password = local.postgres_password

  lifecycle {
    # Ignore password changes after initial creation
    ignore_changes = [password]
  }
}
