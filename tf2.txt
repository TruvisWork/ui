WITH partitioned_tables AS (
    SELECT DISTINCT
        database,
        schema,
        table_name,
        partition_columns  -- Using only partition_columns, NOT partition_text
    FROM partition_metadata
    WHERE partition_columns IS NOT NULL
      AND partition_columns::text != '[]'
      AND partition_columns::text != ''
),

partition_columns_expanded AS (
    SELECT 
        pt.database,
        pt.schema,
        pt.table_name,
        LOWER(TRIM(unnested_col::text)) AS partition_column
    FROM partitioned_tables pt
    CROSS JOIN LATERAL (
        SELECT unnest(string_to_array(pt.partition_columns::text, ',')) AS unnested_col
    ) AS unnested
    WHERE TRIM(unnested_col::text) != ''
),
-- Get all table occurrences from column_info
table_occurrences AS (
    SELECT DISTINCT
        database,
        schema,
        table_name,
        occurrence
    FROM column_info
    WHERE table_name IS NOT NULL
),
-- Get WHERE clause columns for each table occurrence
where_clause_columns AS (
    SELECT DISTINCT
        ci.database,
        ci.schema,
        ci.table_name,
        ci.occurrence,
        LOWER(TRIM(ci.column_name)) AS column_name
    FROM column_info ci
    WHERE ci.clause = 'WHERE'
      AND ci.table_name IS NOT NULL
      AND ci.column_name IS NOT NULL
),
-- Check which table occurrences have partition column filters
occurrences_with_partition_filter AS (
    SELECT DISTINCT
        wcc.database,
        wcc.schema,
        wcc.table_name,
        wcc.occurrence
    FROM where_clause_columns wcc
    INNER JOIN partition_columns_expanded pce
        ON LOWER(TRIM(wcc.table_name)) = LOWER(TRIM(pce.table_name))
        AND wcc.column_name = pce.partition_column
        AND COALESCE(wcc.database, '') = COALESCE(pce.database, '')
        AND COALESCE(wcc.schema, '') = COALESCE(pce.schema, '')
),
-- Find table occurrences that DON'T have partition column filters
occurrences_without_partition_filter AS (
    SELECT 
        to_occ.database,
        to_occ.schema,
        to_occ.table_name,
        to_occ.occurrence
    FROM table_occurrences to_occ
    INNER JOIN partitioned_tables pt
        ON LOWER(TRIM(to_occ.table_name)) = LOWER(TRIM(pt.table_name))
        AND COALESCE(to_occ.database, '') = COALESCE(pt.database, '')
        AND COALESCE(to_occ.schema, '') = COALESCE(pt.schema, '')
    LEFT JOIN occurrences_with_partition_filter owpf
        ON COALESCE(to_occ.database, '') = COALESCE(owpf.database, '')
        AND COALESCE(to_occ.schema, '') = COALESCE(owpf.schema, '')
        AND to_occ.table_name = owpf.table_name
        AND to_occ.occurrence = owpf.occurrence
    WHERE owpf.occurrence IS NULL
)
-- Final result: Tables that have at least one occurrence without partition column filter
SELECT DISTINCT
    database,
    schema,
    table_name,
    COUNT(DISTINCT occurrence) AS occurrences_without_partition_filter,
    STRING_AGG(DISTINCT occurrence::TEXT, ', ' ORDER BY occurrence::TEXT) AS missing_filter_occurrences
FROM occurrences_without_partition_filter
GROUP BY database, schema, table_name
ORDER BY database, schema, table_name;
