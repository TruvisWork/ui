"""
================================================================================
BigQuery Slot Usage Analysis
================================================================================

TECHNICAL DOCUMENTATION (For Developers)
=========================================

DATA SOURCE
-----------
This script uses data from BigQuery's INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT.

The key columns are:
  - period_start: Timestamp for each 1-second measurement period
  - period_slot_ms: Slot-milliseconds consumed during that 1-second period

SQL QUERY TO EXTRACT DATA:
--------------------------
SELECT
  period_start,
  period_slot_ms
FROM
  `region-us`.INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT
WHERE
  period_start BETWEEN TIMESTAMP("2024-01-01") AND TIMESTAMP("2024-01-31")
  AND job_type = "QUERY"
ORDER BY period_start

TIMEFRAME SELECTION
-------------------
- Recommended: 7-30 days of data for accurate patterns
- Include weekdays AND weekends to capture full usage cycles
- Avoid holiday periods unless representative of normal usage

WHY THIS DATA SOURCE?
--------------------
JOBS_TIMELINE provides PER-SECOND slot measurements (true concurrency).
Alternative: JOBS_BY_PROJECT gives aggregate slot_seconds per query, 
but requires estimating concurrency via: avg_slots = slot_seconds / duration_seconds
This estimation assumes uniform slot usage, which is often inaccurate.

CALCULATION METHODOLOGY
-----------------------

1. CONCURRENT SLOTS (at each second):
   concurrent_slots = period_slot_ms / 1000
   
   Explanation: period_slot_ms is in milliseconds. Dividing by 1000 converts
   to slot-seconds. Since the measurement period is exactly 1 second,
   this gives the average number of concurrent slots during that second.
   
   Example: If period_slot_ms = 5000, then 5000/1000 = 5 concurrent slots

2. PERCENTILES:
   - p50 (median): 50% of time usage is below this value
   - p90: 90% of time usage is below this value  
   - p95: 95% of time usage is below this value
   - p99: 99% of time usage is below this value
   
   These are calculated using numpy.percentile() on all concurrent_slots values.

3. AVERAGE:
   Simple arithmetic mean of all concurrent_slots values over the time period.

================================================================================
"""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime

# ==========================
# CONFIG
# ==========================
CSV_FILE = "input.csv"
OUTPUT_PREFIX = "slot_analysis"

# ==========================
# LOAD DATA
# ==========================
print("="*70)
print("BIGQUERY SLOT USAGE ANALYSIS")
print("="*70)

df = pd.read_csv(CSV_FILE)

# Validate and process timeline data
if not {"period_start", "period_slot_ms"}.issubset(df.columns):
    raise ValueError("Expected columns: period_start, period_slot_ms")

df["period_start"] = pd.to_datetime(df["period_start"])
df["concurrent_slots"] = df["period_slot_ms"] / 1000

# Aggregate by time period (sum all slots at each timestamp)
concurrent = (
    df.groupby("period_start")["concurrent_slots"]
    .sum()
    .reset_index()
    .rename(columns={"period_start": "time"})
)

# ==========================
# TECHNICAL METRICS
# ==========================
avg_slots = concurrent["concurrent_slots"].mean()
p50 = np.percentile(concurrent["concurrent_slots"], 50)
p75 = np.percentile(concurrent["concurrent_slots"], 75)
p90 = np.percentile(concurrent["concurrent_slots"], 90)
p95 = np.percentile(concurrent["concurrent_slots"], 95)
p99 = np.percentile(concurrent["concurrent_slots"], 99)
peak = concurrent["concurrent_slots"].max()
min_slots = concurrent["concurrent_slots"].min()

# Time analysis
start_time = concurrent["time"].min()
end_time = concurrent["time"].max()
duration_hours = (end_time - start_time).total_seconds() / 3600
total_data_points = len(concurrent)

# ==========================
# TECHNICAL OUTPUT
# ==========================
print("\n" + "-"*70)
print("TECHNICAL SECTION (For Developers)")
print("-"*70)
print(f"\nDATA TIMEFRAME:")
print(f"  Start: {start_time}")
print(f"  End:   {end_time}")
print(f"  Duration: {duration_hours:.1f} hours ({duration_hours/24:.1f} days)")
print(f"  Data points: {total_data_points:,}")

print(f"\nCALCULATION METHOD:")
print(f"  concurrent_slots = period_slot_ms / 1000")
print(f"  (Each period_start represents a 1-second window)")

print(f"\nRAW METRICS:")
print(f"  Minimum:  {min_slots:>10.1f} slots")
print(f"  Average:  {avg_slots:>10.1f} slots")
print(f"  p50:      {p50:>10.1f} slots")
print(f"  p75:      {p75:>10.1f} slots")
print(f"  p90:      {p90:>10.1f} slots")
print(f"  p95:      {p95:>10.1f} slots")
print(f"  p99:      {p99:>10.1f} slots")
print(f"  Peak:     {peak:>10.1f} slots")

# ==========================
# BUSINESS OUTPUT
# ==========================
print("\n" + "-"*70)
print("BUSINESS SECTION (For Stakeholders)")
print("-"*70)

print("""
WHAT ARE PERCENTILES?
---------------------
Percentiles tell you how often your usage stays below a certain level:

  • p50 = MEDIAN: Half the time you use MORE, half the time you use LESS
  • p90 = 90% of the time, your usage is below this number
  • p95 = 95% of the time, your usage is below this number
  • p99 = 99% of the time, your usage is below this number (rare spikes above)

PRICING STRATEGY:
-----------------
""")

print(f"  📦 RESERVED SLOTS (baseline commitment)")
print(f"     Recommendation: {avg_slots:.0f} - {p75:.0f} slots")
print(f"     → Cheapest per-slot rate, pay 24/7")
print(f"     → Set at average or p75 to cover ~50-75% of time")

print(f"\n  ⚡ FLEX SLOTS (hourly commitment)")
print(f"     Recommendation: Up to {p95:.0f} slots capacity")
print(f"     → Buy hourly when usage exceeds reserved")
print(f"     → Covers the {100-75:.0f}%-{100-5:.0f}% usage zone")

print(f"\n  💰 ON-DEMAND (no commitment)")
print(f"     Peak needs: {peak:.0f} slots")
print(f"     → Most expensive, but no commitment")
print(f"     → Only for rare spikes above p95")

# ==========================
# GRAPH 1: Timeline with Percentile Lines
# ==========================
fig, axes = plt.subplots(2, 2, figsize=(18, 12))

# Graph 1: Full Timeline
ax1 = axes[0, 0]
ax1.fill_between(concurrent["time"], concurrent["concurrent_slots"], 
                 alpha=0.3, color='blue')
ax1.plot(concurrent["time"], concurrent["concurrent_slots"], 
         linewidth=0.5, alpha=0.7, color='blue')

# Add percentile lines
ax1.axhline(avg_slots, color='green', linestyle='-', linewidth=2, 
            label=f'Average = {avg_slots:.0f} slots')
ax1.axhline(p50, color='yellow', linestyle='--', linewidth=1.5, 
            label=f'p50 (Median) = {p50:.0f} slots')
ax1.axhline(p90, color='orange', linestyle='--', linewidth=1.5, 
            label=f'p90 = {p90:.0f} slots')
ax1.axhline(p95, color='red', linestyle='--', linewidth=2, 
            label=f'p95 = {p95:.0f} slots')
ax1.axhline(peak, color='darkred', linestyle=':', linewidth=1, 
            label=f'Peak = {peak:.0f} slots')

ax1.set_title("Concurrent Slot Usage Over Time\n(with Percentile Reference Lines)", fontsize=12, fontweight='bold')
ax1.set_xlabel("Time")
ax1.set_ylabel("Concurrent Slots")
ax1.legend(loc='upper right', fontsize=9)
ax1.grid(alpha=0.3)

# Graph 2: Distribution Histogram
ax2 = axes[0, 1]
ax2.hist(concurrent["concurrent_slots"], bins=50, alpha=0.7, color='steelblue', edgecolor='white')
ax2.axvline(avg_slots, color='green', linestyle='-', linewidth=2, label=f'Average = {avg_slots:.0f}')
ax2.axvline(p50, color='yellow', linestyle='--', linewidth=2, label=f'p50 = {p50:.0f}')
ax2.axvline(p90, color='orange', linestyle='--', linewidth=2, label=f'p90 = {p90:.0f}')
ax2.axvline(p95, color='red', linestyle='--', linewidth=2, label=f'p95 = {p95:.0f}')
ax2.axvline(peak, color='darkred', linestyle=':', linewidth=2, label=f'Peak = {peak:.0f}')

ax2.set_title("Distribution of Slot Usage\n(How often each usage level occurs)", fontsize=12, fontweight='bold')
ax2.set_xlabel("Concurrent Slots")
ax2.set_ylabel("Frequency (count of time periods)")
ax2.legend(loc='upper right', fontsize=9)
ax2.grid(alpha=0.3)

# Graph 3: Pricing Zones
ax3 = axes[1, 0]
ax3.fill_between(concurrent["time"], 0, avg_slots, 
                 alpha=0.4, color='green', label=f'RESERVED Zone (0-{avg_slots:.0f} slots)')
ax3.fill_between(concurrent["time"], avg_slots, 
                 np.minimum(concurrent["concurrent_slots"], p95),
                 where=(concurrent["concurrent_slots"] > avg_slots),
                 alpha=0.4, color='orange', label=f'FLEX Zone ({avg_slots:.0f}-{p95:.0f} slots)')
ax3.fill_between(concurrent["time"], p95, concurrent["concurrent_slots"],
                 where=(concurrent["concurrent_slots"] > p95),
                 alpha=0.5, color='red', label=f'ON-DEMAND Zone (>{p95:.0f} slots)')
ax3.plot(concurrent["time"], concurrent["concurrent_slots"], 
         linewidth=0.5, alpha=0.5, color='black')

ax3.set_title("Pricing Strategy Zones\n(Reserved = Green, Flex = Orange, On-Demand = Red)", fontsize=12, fontweight='bold')
ax3.set_xlabel("Time")
ax3.set_ylabel("Concurrent Slots")
ax3.legend(loc='upper right', fontsize=9)
ax3.grid(alpha=0.3)

# Graph 4: 24-Hour Profile
ax4 = axes[1, 1]
concurrent["hour"] = concurrent["time"].dt.hour
hourly_avg = concurrent.groupby("hour")["concurrent_slots"].mean()
hourly_p95 = concurrent.groupby("hour")["concurrent_slots"].quantile(0.95)

ax4.fill_between(hourly_avg.index, hourly_avg.values, alpha=0.4, color='blue')
ax4.plot(hourly_avg.index, hourly_avg.values, linewidth=2, color='blue', marker='o', label='Average per Hour')
ax4.plot(hourly_p95.index, hourly_p95.values, linewidth=2, color='red', marker='s', linestyle='--', label='p95 per Hour')
ax4.axhline(avg_slots, color='green', linestyle='-', linewidth=1, alpha=0.5)

ax4.set_title("24-Hour Usage Profile\n(Average slots by hour of day)", fontsize=12, fontweight='bold')
ax4.set_xlabel("Hour of Day (0-23)")
ax4.set_ylabel("Concurrent Slots")
ax4.set_xticks(range(0, 24))
ax4.legend(loc='upper right', fontsize=9)
ax4.grid(alpha=0.3)

plt.tight_layout()
plt.savefig(f"{OUTPUT_PREFIX}_dashboard.png", dpi=150, bbox_inches='tight')
print(f"\n✅ Saved: {OUTPUT_PREFIX}_dashboard.png")
plt.show()

# ==========================
# SUMMARY TABLE
# ==========================
print("\n" + "="*70)
print("RECOMMENDATION SUMMARY")
print("="*70)
print(f"""
┌─────────────────┬─────────────┬──────────────────────────────────────────┐
│ Pricing Tier    │ Slots       │ When to Use                              │
├─────────────────┼─────────────┼──────────────────────────────────────────┤
│ RESERVED        │ {avg_slots:>6.0f}      │ Baseline commitment (always running)     │
│ FLEX            │ {p95-avg_slots:>6.0f}      │ Additional capacity for peak hours       │
│ ON-DEMAND       │ {peak-p95:>6.0f}      │ Rare spikes only (~{100-95:.0f}% of time)              │
├─────────────────┼─────────────┼──────────────────────────────────────────┤
│ TOTAL CAPACITY  │ {peak:>6.0f}      │ Maximum observed usage                   │
└─────────────────┴─────────────┴──────────────────────────────────────────┘

Graph Guide:
• Timeline Graph: Shows actual usage patterns over your data period
• Distribution Graph: Shows how often each slot level occurs
• Pricing Zones Graph: Visualizes where each pricing tier applies
• 24-Hour Profile: Shows daily patterns to identify peak business hours
""")
