stage("Install Gradle") {
    steps {
        script {
            sh '''
            mkdir -p /opt/gradle
            curl -u $NEXUS_USER:$NEXUS_PASS \
              -o /opt/gradle/gradle-8.3.zip \
              "https://nexus.***.com/repository/tools/gradle/8.3/gradle-8.3.zip"

            unzip -qo /opt/gradle/gradle-8.3.zip -d /opt/gradle
            export GRADLE_HOME=/opt/gradle/gradle-8.3
            export PATH=$GRADLE_HOME/bin:$PATH

            gradle -v
            '''
        }
    }
}
