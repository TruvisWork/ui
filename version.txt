import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# =========================================================
# 1. LOAD DATA (3 FILES)
# =========================================================

files = {
    'procs': 'procs.csv',
    'user_fired': 'user_fired.csv',
    'sa_non_proc': 'sa_non_proc.csv'
}

dfs = []
for qtype, path in files.items():
    df = pd.read_csv(path)
    df['query_type'] = qtype
    dfs.append(df)

df = pd.concat(dfs, ignore_index=True)

# Convert to datetime
df['start_time'] = pd.to_datetime(df['start_time'])
df['end_time'] = pd.to_datetime(df['end_time'])

# Duration & average slot rate
df['duration_seconds'] = (
    df['end_time'] - df['start_time']
).dt.total_seconds().clip(lower=1)

df['slots_per_second'] = df['slot_seconds'] / df['duration_seconds']

print(f"Loaded {len(df):,} queries")
print(f"Time range: {df['start_time'].min()} → {df['end_time'].max()}")

# =========================================================
# 2. BUILD CONCURRENT SLOT TIMELINE (5-MIN RESOLUTION)
# =========================================================

def compute_concurrency(df, freq='5min'):
    start = df['start_time'].min().floor(freq)
    end = df['end_time'].max().ceil(freq)

    timeline = pd.date_range(start, end, freq=freq)
    rows = []

    for t0, t1 in zip(timeline[:-1], timeline[1:]):
        running = df[
            (df['start_time'] < t1) &
            (df['end_time'] > t0)
        ]

        rows.append({
            'time': t0 + (t1 - t0)/2,
            'total_slots': running['slots_per_second'].sum(),
            'procs': running.loc[running['query_type']=='procs','slots_per_second'].sum(),
            'user_fired': running.loc[running['query_type']=='user_fired','slots_per_second'].sum(),
            'sa_non_proc': running.loc[running['query_type']=='sa_non_proc','slots_per_second'].sum(),
            'query_count': len(running)
        })

    return pd.DataFrame(rows)

timeline = compute_concurrency(df, freq='5min')

# =========================================================
# 3. PEAK CONCURRENT SLOT USAGE
# =========================================================

peak = timeline.loc[timeline['total_slots'].idxmax()]

print("\n" + "="*60)
print("PEAK CONCURRENT SLOT USAGE")
print("="*60)
print(f"Time                : {peak['time']}")
print(f"Total slots         : {peak['total_slots']:.2f}")
print(f"Concurrent queries  : {int(peak['query_count'])}")
print("="*60)

# =========================================================
# 4. 24-HOUR AVERAGE PROFILE (CLEAN VIEW)
# =========================================================

timeline['hour'] = timeline['time'].dt.hour

hourly_avg = (
    timeline
    .groupby('hour')[['total_slots','procs','user_fired','sa_non_proc']]
    .mean()
    .reset_index()
)

# =========================================================
# 5. PLOTS
# =========================================================

# --- A. Full timeline with peak marker ---
plt.figure(figsize=(16,6))
plt.plot(timeline['time'], timeline['total_slots'], label='Total Slots', linewidth=1.5)

plt.scatter(
    peak['time'],
    peak['total_slots'],
    color='red',
    zorder=5,
    label=f"Peak {peak['total_slots']:.1f} slots"
)

plt.title('Concurrent BigQuery Slot Usage – 30 Days')
plt.xlabel('Time')
plt.ylabel('Concurrent Slots')
plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

# --- B. Average 24-hour day (stacked) ---
plt.figure(figsize=(14,6))

plt.stackplot(
    hourly_avg['hour'],
    hourly_avg['procs'],
    hourly_avg['sa_non_proc'],
    hourly_avg['user_fired'],
    labels=['procs','sa_non_proc','user_fired'],
    alpha=0.85
)

plt.xlabel('Hour of Day (0–23)')
plt.ylabel('Average Concurrent Slots')
plt.title('Average BigQuery Slot Usage – 24 Hour Profile (30 Days)')
plt.xticks(range(24))
plt.legend(loc='upper left')
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

# =========================================================
# 6. SUMMARY STATS
# =========================================================

print("\n" + "="*60)
print("SUMMARY STATISTICS")
print("="*60)
print(f"Average concurrent slots : {timeline['total_slots'].mean():.2f}")
print(f"Median concurrent slots  : {timeline['total_slots'].median():.2f}")
print(f"P90 concurrent slots     : {timeline['total_slots'].quantile(0.90):.2f}")
print(f"P95 concurrent slots     : {timeline['total_slots'].quantile(0.95):.2f}")
print("="*60)
