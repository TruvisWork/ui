    /**
     * Sets the catalog to use (allows building on an existing catalog)

    /**
     * Creates a table with the given name and columns in the specified dataset
     */
    public ProcQueriesTestCatalogBuilder withTable(String datasetName, String tableName, List<Column> columns) {
        Identifier datasetId = Identifier.of(datasetName, sqlDialect);
        Identifier tableId = Identifier.of(tableName, sqlDialect);
        List<Identifier> namespace = List.of(dialect, projectName, datasetId);

        catalog.addEntity(new Table(namespace, tableId, columns, System.currentTimeMillis()));
        return this;

    /**
     * Creates a view with the given name, columns, and SQL definition in the specified dataset
     */

        Identifier datasetId = Identifier.of(datasetName, sqlDialect);

    /**
     * Helper method to create a column with the given name, position, and data type
     */
    public static Column createColumn(String datasetName, String tableName, String columnName, int position, DataType dataType) {
        SqlDialect sqlDialect = new BigQuerySqlDialect();
        Identifier dialect = Identifier.of(Product.BIG_QUERY.name, sqlDialect);
        Identifier projectName = Identifier.of("test", sqlDialect);
        Identifier datasetId = Identifier.of(datasetName, sqlDialect);
        Identifier tableId = Identifier.of(tableName, sqlDialect);
        List<Identifier> columnNamespace = List.of(dialect, projectName, datasetId, tableId);

        return Column.builder()
                .namespace(columnNamespace)
                .name(Identifier.of(columnName, sqlDialect))
                .dialect(sqlDialect)
                .ordinalPosition(position)
                .type(dataType)
                .build();
    }

    private void addWorkflowRulesView(Identifier datasetId) {
        // This view is based on rules_audit with unnested workflow.rules
        // The view columns are: rules.* except (name, description),
        // rules.name as description, rules.description as name,
        // workflow.name as workflow_name, workflow.username, workflow.timestamp, etc.
        Identifier viewId = Identifier.of("workflow_rules_vw", sqlDialect);
        List<Identifier> namespace = List.of(dialect, projectName, datasetId);

        List<Column> columns = new ArrayList<>();
        int position = 1;
        List<Identifier> columnNamespace = List.of(dialect, projectName, datasetId, viewId);

        // Add all 22 columns from the view schema
        // 1. actions (STRING, REPEATED)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("actions", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.ARRAY, 0, 0)).build());
        // 2. createdAt (INTEGER, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("createdAt", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.INTEGER, 0, 0)).build());
        // 3. createdBy (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("createdBy", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 4. id (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("id", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 5. originalId (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("originalId", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 6. rulesProjectId (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("rulesProjectId", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 7. rulesProjectSnapshotId (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("rulesProjectSnapshotId", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 8. scoreEnabled (BOOLEAN, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("scoreEnabled", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.BOOLEAN, 0, 0)).build());
        // 9. workflowRuleBlockName (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("workflowRuleBlockName", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 10. targetVarValue (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("targetVarValue", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 11. comment (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("comment", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 500, 0)).build());
        // 12. condition (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("condition", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 1000, 0)).build());
        // 13. rulesProjectDesc (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("rulesProjectDesc", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 500, 0)).build());
        // 14. rulesProjectSnapshotDesc (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("rulesProjectSnapshotDesc", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 500, 0)).build());
        // 15. variables (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("variables", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 1000, 0)).build());
        // 16. description (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("description", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 500, 0)).build());
        // 17. name (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("name", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 500, 0)).build());
        // 18. workflow_name (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("workflow_name", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 19. username (STRING, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("username", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.VARCHAR, 100, 0)).build());
        // 20. timestamp (INTEGER, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("timestamp", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.INTEGER, 0, 0)).build());
        // 21. timestamp_formatted (TIMESTAMP, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("timestamp_formatted", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.TIMESTAMP, 0, 0)).build());
        // 22. updatedTimestamp (TIMESTAMP, NULLABLE)
        columns.add(Column.builder().namespace(columnNamespace).name(Identifier.of("updatedTimestamp", sqlDialect)).dialect(sqlDialect).ordinalPosition(position++).type(DataType.create(SqlTypeName.TIMESTAMP, 0, 0)).build());

        // Add the SQL definition for the view
        String viewSql = "CREATE VIEW amh_fz_fdr_dev_sit.workflow_rules_vw AS SELECT * EXCEPT(row_num) " +
                "FROM ( " +
                "  SELECT " +
                "    rules.* EXCEPT(name, description), " +
                "    rules.name AS description, " +
                "    rules.description AS name, " +
                "    workflow.name AS workflow_name, " +
                "    workflow.username, " +
                "    workflow.timestamp, " +
                "    TIMESTAMP_MILLIS(workflow.timestamp) AS timestamp_formatted, " +
                "    updatedTimestamp, " +
                "    ROW_NUMBER() OVER(PARTITION BY id ORDER BY workflow.timestamp DESC) AS row_num " +
                "  FROM AMH_FZ_FDR_PROD.rules_audit " +
                "  LEFT JOIN UNNEST(workflow.rules) AS rules " +
                ") " +
                "WHERE row_num = 1";

        catalog.addEntity(new View(namespace, viewId, columns, viewSql, System.currentTimeMillis()));
    }
