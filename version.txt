import pandas as pd
import numpy as np

# =========================
# CONFIG
# =========================
CSV_FILE = "queries_timeline.csv"
OUTPUT_FILE = "monthly_slot_metrics.csv"

# =========================
# LOAD DATA
# =========================
df = pd.read_csv(CSV_FILE)

df["period_start"] = pd.to_datetime(df["period_start"])
df["month"] = df["period_start"].dt.to_period("M")

# =========================
# MONTHLY AGGREGATION
# =========================
monthly = []

for month, g in df.groupby("month"):
    slots = g["concurrent_slots"]
    bytes_sec = g["bytes_billed_per_second"]

    monthly.append({
        "month": str(month),

        # SLOT METRICS
        "slots_min": slots.min(),
        "slots_p50": np.percentile(slots, 50),
        "slots_avg": slots.mean(),
        "slots_p75": np.percentile(slots, 75),
        "slots_p90": np.percentile(slots, 90),
        "slots_p95": np.percentile(slots, 95),
        "slots_p99": np.percentile(slots, 99),
        "slots_max": slots.max(),

        # BYTES METRICS
        "avg_bytes_per_sec": bytes_sec.mean(),
        "total_bytes_billed": bytes_sec.sum(),

        # RELATIONSHIP METRIC (for equation fitting)
        "slot_bytes_corr": slots.corr(bytes_sec),

        # VOLUME
        "seconds_observed": len(g)
    })

monthly_df = pd.DataFrame(monthly)

# =========================
# SAVE
# =========================
monthly_df.to_csv(OUTPUT_FILE, index=False)

print("✅ Monthly metrics saved:", OUTPUT_FILE)
print(monthly_df.head())
