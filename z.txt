DO $$
DECLARE
    is_partitioned BOOLEAN;
BEGIN
    SELECT EXISTS (
        SELECT 1
        FROM pg_partitioned_table pt
        JOIN pg_class c ON pt.partrelid = c.oid
        WHERE c.relname = 'sql_statement_info'
    ) INTO is_partitioned;

    IF is_partitioned THEN
        RAISE NOTICE 'Table already partitioned. Skipping.';
    ELSE
        RAISE NOTICE 'Table not partitioned. Creating partitions...';

        -- Step 1: Rename old table
        EXECUTE 'ALTER TABLE sql_statement_info RENAME TO sql_statement_info_old';

        -- Step 2: Create partitioned parent table (modify columns exactly as your table)
        EXECUTE '
            CREATE TABLE sql_statement_info (
                id BIGINT,
                version_id BIGINT,
                statement_hash TEXT,
                statement TEXT,
                execution_time BIGINT,
                created_at TIMESTAMP
            )
            PARTITION BY RANGE (version_id);
        ';

        -- Step 3: Create partitions
        EXECUTE '
            CREATE TABLE sql_statement_info_p0   PARTITION OF sql_statement_info FOR VALUES FROM (0)    TO (100);
            CREATE TABLE sql_statement_info_p1   PARTITION OF sql_statement_info FOR VALUES FROM (100)  TO (200);
            CREATE TABLE sql_statement_info_p2   PARTITION OF sql_statement_info FOR VALUES FROM (200)  TO (300);
            CREATE TABLE sql_statement_info_pmax PARTITION OF sql_statement_info FOR VALUES FROM (300)  TO (MAXVALUE);
        ';

        -- Step 4: Move old data into new partitioned table
        EXECUTE '
            INSERT INTO sql_statement_info
            SELECT * FROM sql_statement_info_old;
        ';

        -- Step 5: Drop old table
        EXECUTE 'DROP TABLE sql_statement_info_old';

        RAISE NOTICE 'Partitioning complete!';
    END IF;
END $$;
