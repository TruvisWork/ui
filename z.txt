terraform {
  required_providers {
    google = {
      source  = "hashicorp/google"
      version = ">= 4.0"
    }
  }
}

# ------------------------------------------------------------------------------
# Locals - Derived Values
# ------------------------------------------------------------------------------
locals {
  env                        = regex(".*-([^-]+)$", var.my_project)[0]
  iam_service_account_emails = [for sa in var.iam_service_account_names : "${sa}@${var.my_project}.iam.gserviceaccount.com"]
  build_account_email        = "${var.build_account_name}@${var.my_project}.iam.gserviceaccount.com"
  vm_sa_email                = "query-genie@${var.my_project}.iam.gserviceaccount.com"
  host_subnet_uri            = "projects/${var.host_project}/regions/${var.region}/subnetworks/${var.host_subnet}"
  def_subnet_uri             = "projects/${var.my_project}/regions/${var.region}/subnetworks/${var.default_network}"
  vm_instance_name           = "${var.vm_name}-${var.inst_postfix}-extraction-test"
  policy_instance_name       = "${var.policy_name}-${var.inst_postfix}"
  sql_instance_name          = "${var.cloudsql_instance_name}-${local.env}"
  zone                       = "${var.region}-${var.zone_postfix}"
  timezone                   = lookup({ "asia-east2" = "Asia/Hong_Kong" }, var.region, "Asia/Hong_Kong")
  startup_script_bucket      = "${var.my_project}-query-genie-reco"
  kms_compute_key            = "projects/${var.kms_project}/locations/${var.region}/keyRings/computeEngine/cryptoKeys/computeEngine"
  kms_sql_key                = "projects/${var.kms_project}/locations/${var.region}/keyRings/sql/cryptoKeys/sqlSharedKey"
}

provider "google" {
  project                     = var.my_project
  region                      = var.region
  impersonate_service_account = local.build_account_email
}

# ------------------------------------------------------------------------------
# Data Sources
# ------------------------------------------------------------------------------
data "google_project" "project" {
  project_id = var.my_project
}

data "google_compute_address" "static_ip" {
  name = var.ip_static_name
}

# ------------------------------------------------------------------------------
# GCS Resources - The startup script for the VM
# ------------------------------------------------------------------------------
resource "google_storage_bucket_object" "startup_script" {
  count  = var.build_vm ? 1 : 0
  name   = var.startup_script_path
  bucket = local.startup_script_bucket
  source = var.startup_script_local_path
}

# ------------------------------------------------------------------------------
# Resource Policy for VM Scheduling
# ------------------------------------------------------------------------------
resource "google_compute_resource_policy" "vm_schedule" {
  count       = var.build_vm ? 1 : 0
  name        = local.policy_instance_name
  region      = var.region
  project     = var.my_project
  description = "Office Hours for ${var.inst_postfix}"

  instance_schedule_policy {
    vm_stop_schedule = var.vm_stop_schedule
    time_zone        = local.timezone
  }
}

# ------------------------------------------------------------------------------
# Cloud SQL Instance - Native Terraform Resource
# ------------------------------------------------------------------------------
resource "google_sql_database_instance" "main_db" {
  count               = var.build_sql ? 1 : 0
  name                = local.sql_instance_name
  project             = var.my_project
  database_version    = "POSTGRES_13"
  region              = var.region
  deletion_protection = false # Set to true for production environments

  settings {
    tier                 = "db-custom-2-8192"
    availability_type    = "REGIONAL"
    disk_type            = "PD_SSD"
    disk_size            = 10
    backup_configuration {
      enabled                      = true
      start_time                   = "04:00"
      point_in_time_recovery_enabled = true
    }
    ip_configuration {
      ipv4_enabled    = false
      private_network = "projects/${var.my_project}/global/networks/${var.default_network}"
    }
    database_flags {
      name  = "cloudsql.iam_authentication"
      value = "on"
    }
  }
  encryption_key_name = local.kms_sql_key
}

# ------------------------------------------------------------------------------
# GCE Proxy VM Instance
# ------------------------------------------------------------------------------
resource "google_compute_instance" "proxy_vm" {
  count        = var.build_vm ? 1 : 0
  project      = var.my_project
  zone         = local.zone
  name         = local.vm_instance_name
  machine_type = var.machine_type
  tags         = var.network_tags

  boot_disk {
    initialize_params {
      image = "projects/${var.image_project}/global/images/family/${var.image_family_stage2}"
      size  = var.disk_size_gb
      type  = var.disk_type
    }
    kms_key_self_link = local.kms_compute_key
  }

  network_interface {
    subnetwork = local.host_subnet_uri
    network_ip = data.google_compute_address.static_ip.address
  }
  network_interface {
    subnetwork = local.def_subnet_uri
  }

  service_account {
    email  = local.vm_sa_email
    scopes = var.vm_scopes
  }

  scheduling {
    resource_policies = [google_compute_resource_policy.vm_schedule[0].self_link]
  }

  metadata = {
    startup-script-url  = "gs://${local.startup_script_bucket}/${google_storage_bucket_object.startup_script[0].name}"
    cloudsql_private_ip = var.build_sql ? google_sql_database_instance.main_db[0].private_ip_address : ""
  }

  allow_stopping_for_update = true
  deletion_protection       = false

  lifecycle {
    # This block replaces the 'delete_instance_before_create' script.
    # It will destroy and re-create the VM if the startup script GCS object changes.
    replace_triggered_by = [google_storage_bucket_object.startup_script[0]]
  }
}

# ------------------------------------------------------------------------------
# IAM Binding for VM
# ------------------------------------------------------------------------------
resource "google_compute_instance_iam_member" "vm_stopper" {
  count         = var.build_vm ? 1 : 0
  project       = var.my_project
  zone          = local.zone
  instance_name = google_compute_instance.proxy_vm[0].name
  role          = "organizations/103829057055/roles/cr.instancestopStart"
  member        = "serviceAccount:service-${data.google_project.project.number}@compute-system.iam.gserviceaccount.com"
}

# ------------------------------------------------------------------------------
# Cloud SQL Database
# ------------------------------------------------------------------------------
resource "google_sql_database" "database" {
  count    = var.build_sql ? 1 : 0
  instance = google_sql_database_instance.main_db[0].name
  name     = var.db_name
  project  = var.my_project
}

# ------------------------------------------------------------------------------
# Cloud SQL User
# ------------------------------------------------------------------------------
resource "google_sql_user" "postgres_admin" {
  count    = var.build_sql ? 1 : 0
  instance = google_sql_database_instance.main_db[0].name
  name     = "postgres"
  password = var.postgres_password
  project  = var.my_project
}

# ------------------------------------------------------------------------------
# IAM Users for Cloud SQL
# ------------------------------------------------------------------------------
resource "google_sql_user" "iam_users" {
  for_each = var.build_sql ? toset(var.iam_user_emails) : []
  project  = var.my_project
  instance = google_sql_database_instance.main_db[0].name
  name     = each.key
  type     = "CLOUD_IAM_USER"
}

# ------------------------------------------------------------------------------
# IAM Service Accounts for Cloud SQL
# ------------------------------------------------------------------------------
resource "google_sql_user" "iam_service_accounts" {
  for_each = var.build_sql ? toset(local.iam_service_account_emails) : []
  project  = var.my_project
  instance = google_sql_database_instance.main_db[0].name
  name     = each.key
  type     = "CLOUD_IAM_SERVICE_ACCOUNT"
}

# ------------------------------------------------------------------------------
# Outputs
# ------------------------------------------------------------------------------
output "environment" {
  description = "Current environment"
  value       = local.env
}

output "proxy_vm_primary_ip" {
  description = "The primary internal IP address of the proxy VM."
  value       = var.build_vm ? google_compute_instance.proxy_vm[0].network_interface[0].network_ip : "VM not built."
}

output "proxy_vm_secondary_ip" {
  description = "The secondary internal IP address of the proxy VM."
  value       = var.build_vm ? google_compute_instance.proxy_vm[0].network_interface[1].network_ip : "VM not built."
}

output "cloudsql_instance_connection_name" {
  description = "The connection name for the Cloud SQL instance."
  value       = var.build_sql ? google_sql_database_instance.main_db[0].connection_name : "SQL not built."
}

output "cloudsql_private_ip" {
  description = "The private IP of the Cloud SQL instance."
  value       = var.build_sql ? google_sql_database_instance.main_db[0].private_ip_address : "SQL not built."
}
