import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
from datetime import datetime, timedelta
import numpy as np

# Sample data structure - replace with your actual data
data = {
    'start_time': ['2025-11-21 10:00:00', '2025-11-21 10:05:00', '2025-11-21 10:10:00'],
    'end_time': ['2025-11-21 10:30:00', '2025-11-21 10:15:00', '2025-11-21 10:40:00'],
    'slot_seconds': [3000, 1500, 5000],
    'query_type': ['procs', 'user_fired', 'sa_non_proc']
}

df = pd.DataFrame(data)
print(f"Processing {len(df)} queries...")

# Convert to datetime
df['start_time'] = pd.to_datetime(df['start_time'])
df['end_time'] = pd.to_datetime(df['end_time'])

# Calculate duration and average slots per query
df['duration_seconds'] = (df['end_time'] - df['start_time']).dt.total_seconds()
df['duration_seconds'] = df['duration_seconds'].replace(0, 1)  # Avoid division by zero

# CRITICAL FIX: slots_used is the AVERAGE concurrent slots during query execution
# slot_seconds = average_concurrent_slots × duration
# Therefore: average_concurrent_slots = slot_seconds / duration
df['avg_slots'] = df['slot_seconds'] / df['duration_seconds']

print(f"Average slot usage per query range: {df['avg_slots'].min():.2f} to {df['avg_slots'].max():.2f}")


def calculate_concurrent_slots_fixed(df, bucket_size='15min'):
    """
    Calculate TRUE concurrent slot usage over time
    This counts how many slots are being used at any given moment
    """
    print(f"Using {bucket_size} time buckets...")
    
    # Get time range
    min_time = df['start_time'].min()
    max_time = df['end_time'].max()
    print(f"Time range: {min_time} to {max_time}")
    
    # Create time buckets
    time_range = pd.date_range(
        start=min_time.floor(bucket_size),
        end=max_time.ceil(bucket_size),
        freq=bucket_size
    )
    
    results = []
    
    # For each time bucket, calculate concurrent slot usage
    for i, bucket_start in enumerate(time_range[:-1]):
        bucket_end = time_range[i + 1]
        bucket_mid = bucket_start + (bucket_end - bucket_start) / 2
        
        # Find all queries running during this bucket
        running_queries = df[
            (df['start_time'] < bucket_end) & 
            (df['end_time'] > bucket_start)
        ]
        
        # CRITICAL FIX: Calculate weighted average based on overlap
        slots_by_type = {'procs': 0, 'user_fired': 0, 'sa_non_proc': 0}
        
        bucket_duration = (bucket_end - bucket_start).total_seconds()
        
        for _, query in running_queries.iterrows():
            # Calculate overlap duration
            overlap_start = max(query['start_time'], bucket_start)
            overlap_end = min(query['end_time'], bucket_end)
            overlap_duration = (overlap_end - overlap_start).total_seconds()
            
            # Weight the slot usage by overlap fraction
            # If a query runs for 50% of the bucket, it contributes 50% of its slots
            overlap_fraction = overlap_duration / bucket_duration
            slots_by_type[query['query_type']] += query['avg_slots'] * overlap_fraction
        
        results.append({
            'time': bucket_mid,
            'procs': slots_by_type['procs'],
            'user_fired': slots_by_type['user_fired'],
            'sa_non_proc': slots_by_type['sa_non_proc'],
            'total': sum(slots_by_type.values()),
            'query_count': len(running_queries)
        })
    
    return pd.DataFrame(results)


print("\nCalculating concurrent slot usage...")

# Calculate with different granularities
timeline_15min = calculate_concurrent_slots_fixed(df, '15min')
timeline_4h = calculate_concurrent_slots_fixed(df, '4h')

timeline_df = timeline_15min

print(f"\nGenerated {len(timeline_df)} time points")
print(f"Concurrent slots range: {timeline_df['total'].min():.2f} to {timeline_df['total'].max():.2f}")

# Create visualization

                timeline_df['procs'] + timeline_df['sa_non_proc'],
                label='sa_non_proc', color=colors['sa_non_proc'], alpha=0.7)
ax.fill_between(timeline_df['time'], 
                timeline_df['procs'] + timeline_df['sa_non_proc'],
                timeline_df['total'],
                label='user_fired', color=colors['user_fired'], alpha=0.7)

# Overlay 4-hour average
ax.plot(timeline_4h['time'], timeline_4h['total'],

# Find and mark peak
