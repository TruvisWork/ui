@Slf4j
public class SelectVisitor extends BaseStatementVisitor {

    private final JoinAnalyzer joinAnalyzer;
    private final PartitionColumnFunctionDetector partitionDetector;
    private final Dialect dialect;
    public SelectVisitor(String userName, String defaultDatabase, String defaultSchema, RelationshipExtractor relationshipExtractor, EntityCatalog entityCatalog) {
        super(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
        this.joinAnalyzer = new JoinAnalyzer(entityCatalog);
        this.partitionDetector = new PartitionColumnFunctionDetector(entityCatalog, defaultDatabase, defaultSchema);
        this.dialect = new BigQuerySqlDialect();
    }

    private SqlNodeVisitor.Result currentResult;
    private List<String> externalCteAliases = new ArrayList<>();

    /**
     * Sets CTE aliases from parent WITH statement before processing
     */
    public void setCteAliases(List<String> cteAliases) {
        if (cteAliases != null) {
            this.externalCteAliases = new ArrayList<>(cteAliases);
        }
    }

    @Override
    public SqlNodeVisitor.Result visit(SqlCall call) {
        SqlSelect select = (SqlSelect) call;
        SqlNodeVisitor.Result result = new SqlNodeVisitor.Result();
        this.currentResult = result; // Store current result for CTE checking
        
        // Copy external CTE aliases to result context
//        for (String cteAlias : externalCteAliases) {
//            result.getContext().addCteAlias(cteAlias);
//        }
        
        result.setStatementType(StatementType.SELECT);

        SelectClause selectClause = new SelectClause();
        selectClause.setHasSelectAll(isSelectAll(select));
        selectClause.setHasDistinct(select.isDistinct());
        result.getContext().setSelectClause(selectClause);

        if (select.getFrom() != null) {
            var from = select.getFrom();
            log.debug("SelectVisitor.visit: Processing FROM clause, type: {}", from.getClass().getSimpleName());

            if (from instanceof SqlBasicCall basicCall) {
                if (!basicCall.getOperandList().isEmpty() && basicCall.getOperandList().get(0) instanceof SqlOrderBy) {
                    OrderByClause orderByClause = new OrderByClause();
                    orderByClause.setIsOderByInsideSubQuery(true);
                    result.getContext().setOrderByClause(orderByClause);
                } else if (basicCall.getOperator().getName().equalsIgnoreCase("AS") &&
                        !basicCall.getOperandList().isEmpty()) {
                    SqlNode firstOperand = basicCall.getOperandList().get(0);
                    log.debug("SelectVisitor.visit: FROM is AS call, first operand type: {}", firstOperand.getClass().getSimpleName());
                    if (firstOperand instanceof SqlIdentifier tableId) {
                        // Skip CTEs from being added as source tables
                        String tableName = getTableName(tableId);
                        if (tableName == null /*|| !result.getContext().getCteAliases().contains(tableName)*/) {
                            result.getSourceTables().add(tableId);
                            log.debug("SelectVisitor.visit: Added table to sourceTables: {}", String.join(".", tableId.names));
                        }
                    } else if (firstOperand instanceof SqlSelect) {
                        // Handle subquery with alias: (SELECT ...) AS alias
                        // Tables from the subquery will be extracted via recursive visitor call below
                        log.debug("SelectVisitor.visit: FROM is AS with subquery, will extract via recursive visitor");
                    }
                }
            } else if (from instanceof SqlIdentifier id) {
                // Skip CTEs from being added as source tables
                String tableName = getTableName(id);
                if (tableName == null /*|| !result.getContext().getCteAliases().contains(tableName)*/) {
                    result.getSourceTables().add(id);
                    log.debug("SelectVisitor.visit: Added table identifier to sourceTables: {}", String.join(".", id.names));
                }
            } else if (from instanceof SqlJoin join) {
                log.debug("SelectVisitor.visit: FROM is JOIN, processing join");
                processJoin(join, result);
            } else if (from instanceof SqlSelect) {
                // Handle subquery directly in FROM clause (without alias)
                // Tables from the subquery will be extracted via recursive visitor call below
                log.debug("SelectVisitor.visit: FROM is direct subquery, will extract via recursive visitor");
            }

            // Process other from clause aspects - this recursively processes subqueries and extracts their tables
            int beforeRecursiveTables = result.getSourceTables().size();
            log.debug("SelectVisitor.visit: Before recursive visitor, sourceTables count: {}", beforeRecursiveTables);
            SqlNodeVisitor visitor = new SqlNodeVisitor(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
            SqlNodeVisitor.Result fromResult = from.accept(visitor);
            if (fromResult != null) {
                log.debug("SelectVisitor.visit: Recursive visitor result has {} sourceTables: {}", 
                        fromResult.getSourceTables().size(),
                        fromResult.getSourceTables().stream()
                                .map(id -> id != null && id.names != null ? String.join(".", id.names) : "null")
                                .toList());
                SqlNodeVisitor.mergeResults(result, fromResult);
                log.debug("SelectVisitor.visit: After merging recursive result, sourceTables count: {}", result.getSourceTables().size());
            } else {
                log.debug("SelectVisitor.visit: Recursive visitor result is null");
            }
            
            // Also explicitly process nested subqueries if FROM is AS with subquery
            // Note: This is in addition to the recursive from.accept(visitor) call above
            // to ensure deeply nested subqueries are fully processed
            if (from instanceof SqlBasicCall fromCall &&
                    fromCall.getOperator().getName().equalsIgnoreCase("AS") &&
                    !fromCall.getOperandList().isEmpty()) {
                SqlNode firstOperand = fromCall.getOperandList().get(0);
                if (firstOperand instanceof SqlSelect nestedSelect) {
                    log.debug("SelectVisitor.visit: Explicitly processing nested subquery in FROM AS");
                    SqlNodeVisitor.Result nestedResult = nestedSelect.accept(visitor);
                    if (nestedResult != null) {
                        log.debug("SelectVisitor.visit: Nested subquery result has {} sourceTables", nestedResult.getSourceTables().size());
                        SqlNodeVisitor.mergeResults(result, nestedResult);
                    }
                }
            }
            
            addAccess(from, result);
        }
        registerTablesForPartitionDetection(result);

        // Initialize WhereClause regardless of WHERE condition existence
        WhereClause whereClause = new WhereClause();
        whereClause.setHasWhereClause(select.getWhere() != null);

        WindowFunctionDetector windowDetector = new WindowFunctionDetector();
        boolean hasUnusedWindowFunction = windowDetector.hasUnusedWindowFunction(
            select.getSelectList(),
            select.getWhere()
        );
        whereClause.setHasUnusedWindowFunction(hasUnusedWindowFunction);

        // Check SELECT list for case-insensitive comparisons
        boolean hasCaseInsensitive = false;
        for (SqlNode node : select.getSelectList()) {
            if (node instanceof SqlCase caseCall) {
                hasCaseInsensitive |= checkCaseExpressionForInsensitiveComparison(caseCall);
            } else if (node instanceof SqlBasicCall basicCall && basicCall.getOperator().getName().equalsIgnoreCase("AS")) {
                // Check if the first operand is a CASE expression
                SqlNode firstOperand = basicCall.getOperandList().get(0);
                if (firstOperand instanceof SqlCase caseCall) {
                    hasCaseInsensitive |= checkCaseExpressionForInsensitiveComparison(caseCall);
                }
            }
        }

        // If WHERE exists, check it too
        if (select.getWhere() != null) {
            SqlNode whereExpr = select.getWhere();
            InClauseAnalyzer.InClauseInfo inInfo = InClauseAnalyzer.analyze(whereExpr);
            whereClause.setHasInWithSubquery(inInfo.hasInWithSubquery());
            whereClause.setHasInWithConstant(inInfo.hasInWithConstant());
            hasCaseInsensitive |= SqlConditionUtils.hasCaseInsensitiveComparison(whereExpr);
            boolean hasFunctionOnPartition = partitionDetector.hasFunctionOnPartitionColumn(whereExpr);
            whereClause.setHasFunctionOnPartitionColumn(hasFunctionOnPartition);

            SqlNodeVisitor visitor = new SqlNodeVisitor(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
            SqlNodeVisitor.Result whereResult = whereExpr.accept(visitor);

            int mainQueryTableCount = result.getSourceTables().size();
            SqlNodeVisitor.mergeResults(result, whereResult);
            result.getColumnInfos().removeIf(ci -> ci.getClause() == ClauseType.WHERE);

            Map<String, Integer> mainQueryOccurrenceMap = new HashMap<>();
            for (int i = 0; i < mainQueryTableCount; i++) {
                SqlIdentifier tableId = result.getSourceTables().get(i);
                String tableName = getTableNameFromEntity(tableId);
                if (tableName == null) {
                    tableName = getTableName(tableId);
                }
                if (tableName != null) {
                    mainQueryOccurrenceMap.put(tableName.toLowerCase(), 1);
                }
            }
            List<ColumnInfo> mainQueryColumns = extractColumnsExcludingSubqueries(whereExpr, ClauseType.WHERE, dialect.product.name, mainQueryOccurrenceMap);
            result.getColumnInfos().addAll(mainQueryColumns);

            Map<String, Integer> subqueryOccurrenceMap = new HashMap<>();
            Map<String, Boolean> mainQueryTableMap = new HashMap<>();
            for (int i = 0; i < mainQueryTableCount; i++) {
                SqlIdentifier tableId = result.getSourceTables().get(i);
                String tableName = getTableNameFromEntity(tableId);
                if (tableName == null) {
                    tableName = getTableName(tableId);
                }
                if (tableName != null) {
                    mainQueryTableMap.put(tableName.toLowerCase(), true);
                }
            }

            for (int i = mainQueryTableCount; i < result.getSourceTables().size(); i++) {
                SqlIdentifier tableId = result.getSourceTables().get(i);
                String tableName = getTableNameFromEntity(tableId);
                if (tableName == null) {
                    tableName = getTableName(tableId);
                }
                if (tableName != null) {
                    String normalizedTableName = tableName.toLowerCase();
                    if (mainQueryTableMap.containsKey(normalizedTableName)) {
                        subqueryOccurrenceMap.put(normalizedTableName, 2);
                    } else {
                        subqueryOccurrenceMap.put(normalizedTableName, 1);
                    }
                }
            }

            List<ColumnInfo> subqueryColumns = extractColumnsFromSubqueries(whereExpr, ClauseType.WHERE, dialect.product.name, subqueryOccurrenceMap, mainQueryTableCount);
            result.getColumnInfos().addAll(subqueryColumns);
        }

        whereClause.setHasCaseInsensitiveComparison(hasCaseInsensitive);
        result.getContext().setWhereClause(whereClause);

        // Extract columns from GROUP BY clause (handles ordinals and expressions)
        if (select.getGroup() != null) {
            Map<String, Integer> tableOccurrenceMap = computeTableOccurrenceMap(result.getSourceTables());
            result.getColumnInfos().addAll(
                ColumnExtractor.extractGroupByColumns(select.getGroup(), select.getSelectList(), dialect.product.name, tableOccurrenceMap)
            );
        }
        partitionDetector.clearRegistry();
        return result;
    }


    private static boolean isSelectAll(SqlSelect select) {
        return select.getSelectList().size() == 1 && "*".equals(select.getSelectList().get(0).toString());
    }

    private void processJoin(SqlJoin join, SqlNodeVisitor.Result result) {
        log.debug("processJoin: Processing join, left type: {}, right type: {}", 
                join.getLeft().getClass().getSimpleName(), 
                join.getRight().getClass().getSimpleName());
        
        // Register tables with JoinAnalyzer
        SqlNode leftTable;
        if (join.getLeft() instanceof SqlJoin leftJoin) {
            processJoin(leftJoin, result);
            leftTable = leftJoin.getRight(); // For a nested join, use its right table as our left table
        } else {
            registerJoinTables(join.getLeft());
            leftTable = join.getLeft();
        }
        registerJoinTables(join.getRight());

        // Recursively process join operands to extract tables from subqueries
        // This handles subqueries at any nesting level within JOIN operands
        SqlNodeVisitor visitor = new SqlNodeVisitor(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
        
        int beforeLeftTables = result.getSourceTables().size();
        log.debug("processJoin: Before processing left operand, sourceTables count: {}", beforeLeftTables);
        
        // Process left operand - extract all tables from subqueries
        SqlNodeVisitor.Result leftResult = join.getLeft().accept(visitor);
        if (leftResult != null) {
            log.debug("processJoin: Left result has {} sourceTables: {}", 
                    leftResult.getSourceTables().size(),
                    leftResult.getSourceTables().stream()
                            .map(id -> id != null && id.names != null ? String.join(".", id.names) : "null")
                            .toList());
            SqlNodeVisitor.mergeResults(result, leftResult);
            log.debug("processJoin: After merging left result, sourceTables count: {}", result.getSourceTables().size());
        } else {
            log.debug("processJoin: Left result is null");
        }
        
        // Also explicitly handle left operand if it's a subquery wrapped in AS
        if (join.getLeft() instanceof SqlBasicCall leftCall &&
                leftCall.getOperator().getName().equalsIgnoreCase("AS") &&
                !leftCall.getOperandList().isEmpty()) {
            SqlNode leftOperand = leftCall.getOperandList().get(0);
            log.debug("processJoin: Left is AS call, first operand type: {}", leftOperand.getClass().getSimpleName());
            if (leftOperand instanceof SqlSelect) {
                log.debug("processJoin: Explicitly processing left subquery in AS");
                SqlNodeVisitor.Result leftSubqueryResult = leftOperand.accept(visitor);
                if (leftSubqueryResult != null) {
                    log.debug("processJoin: Left subquery result has {} sourceTables", leftSubqueryResult.getSourceTables().size());
                    SqlNodeVisitor.mergeResults(result, leftSubqueryResult);
                }
            }
        }
        
        int beforeRightTables = result.getSourceTables().size();
        log.debug("processJoin: Before processing right operand, sourceTables count: {}", beforeRightTables);
        
        // Process right operand - extract all tables from subqueries
        SqlNodeVisitor.Result rightResult = join.getRight().accept(visitor);
        if (rightResult != null) {
            log.debug("processJoin: Right result has {} sourceTables: {}", 
                    rightResult.getSourceTables().size(),
                    rightResult.getSourceTables().stream()
                            .map(id -> id != null && id.names != null ? String.join(".", id.names) : "null")
                            .toList());
            SqlNodeVisitor.mergeResults(result, rightResult);
            log.debug("processJoin: After merging right result, sourceTables count: {}", result.getSourceTables().size());
        } else {
            log.debug("processJoin: Right result is null");
        }
        
        // Also explicitly handle right operand if it's a subquery wrapped in AS
        if (join.getRight() instanceof SqlBasicCall rightCall &&
                rightCall.getOperator().getName().equalsIgnoreCase("AS") &&
                !rightCall.getOperandList().isEmpty()) {
            SqlNode rightOperand = rightCall.getOperandList().get(0);
            log.debug("processJoin: Right is AS call, first operand type: {}", rightOperand.getClass().getSimpleName());
            if (rightOperand instanceof SqlSelect) {
                log.debug("processJoin: Explicitly processing right subquery in AS");
                SqlNodeVisitor.Result rightSubqueryResult = rightOperand.accept(visitor);
                if (rightSubqueryResult != null) {
                    log.debug("processJoin: Right subquery result has {} sourceTables", rightSubqueryResult.getSourceTables().size());
                    SqlNodeVisitor.mergeResults(result, rightSubqueryResult);
                }
            }
        }
        
        log.debug("processJoin: Final sourceTables count: {}", result.getSourceTables().size());

        SqlIdentifier leftIdentifier = getJoinTableIdentifier(leftTable);
        SqlIdentifier rightIdentifier = getJoinTableIdentifier(join.getRight());
        if (leftIdentifier != null && rightIdentifier != null) {
            relationshipExtractor.createJoin(leftIdentifier, rightIdentifier, defaultDatabase, defaultSchema, result.getEntityRelationships());
        }

        if (join.getCondition() != null) {
            JoinClause joinClause = new JoinClause();
            joinClause.setJoinType(join.getJoinType());
            joinClause.setJoinCondition(join.getCondition().toString());
            joinClause.setHasJoinOnStringColumn(joinAnalyzer.isJoinOnStringColumn(join.getCondition()));
            // Populate per-column details used in the join condition
            joinClause.setJoinColumns(joinAnalyzer.extractJoinColumns(join.getCondition()));

            if (leftTable instanceof SqlIdentifier leftId) {
                joinClause.setLeftTable(leftId.toString());
            } else if (leftTable instanceof SqlBasicCall leftCall &&
                    !leftCall.getOperandList().isEmpty() && leftCall.getOperandList().get(0) instanceof SqlIdentifier leftId) {
                joinClause.setLeftTable(leftId.toString()); // Handle "AS" aliases
            }

            if (join.getRight() instanceof SqlIdentifier rightId) {
                joinClause.setRightTable(rightId.toString());
            } else if (join.getRight() instanceof SqlBasicCall rightCall &&
                    !rightCall.getOperandList().isEmpty() && rightCall.getOperandList().get(0) instanceof SqlIdentifier rightId) {
                joinClause.setRightTable(rightId.toString()); // Handle "AS" aliases
            }

            Map<String, Integer> tableOccurrenceMap = computeTableOccurrenceMap(result.getSourceTables());
            result.getColumnInfos().addAll(ColumnExtractor.extractColumns(join.getCondition(), ClauseType.JOIN, dialect.product.name, tableOccurrenceMap));

//            result.getContext().getJoinClauses().add(joinClause);
        }

        addSourceTable(join.getLeft(), result);
        addSourceTable(join.getRight(), result);
    }

    private void registerJoinTables(SqlNode node) {
        if (node instanceof SqlIdentifier id) {
            List<String> qualifiers = new ArrayList<>(id.names);
            String alias = id.names.get(id.names.size() - 1);
            
            // Skip if this is a CTE reference
            if (currentResult != null /*&& currentResult.getContext().getCteAliases().contains(alias)*/) {
                return;
            }
            
            EntityQualifier qualifier = new EntityQualifier(qualifiers, List.of(defaultDatabase, defaultSchema), dialect);
            joinAnalyzer.registerTableAlias(alias, qualifier);
        } else if (node instanceof SqlBasicCall call &&
                call.getOperator().getName().equalsIgnoreCase("AS")) {
            if (call.getOperandList().get(0) instanceof SqlIdentifier tableId &&
                    call.getOperandList().get(1) instanceof SqlIdentifier aliasId) {
                String tableName = tableId.names.get(tableId.names.size() - 1);
                
                // Skip if this is a CTE reference
                if (currentResult != null /*&& currentResult.getContext().getCteAliases().contains(tableName)*/) {
                    return;
                }
                
                List<String> qualifiers = new ArrayList<>(tableId.names);
                EntityQualifier qualifier = new EntityQualifier(qualifiers, List.of(defaultDatabase, defaultSchema), dialect);
                String alias;
                if (aliasId.names.size() == 1) {
                    alias = aliasId.getSimple();
                } else {
                    alias = aliasId.names.get(aliasId.names.size() - 1);
                }
                joinAnalyzer.registerTableAlias(alias, qualifier);
            }
        }
    }

    private SqlIdentifier getJoinTableIdentifier(SqlNode node) {
        if (node instanceof SqlIdentifier id) {
            return id;
        }
        if (node instanceof SqlBasicCall call &&
                call.getOperator().getName().equalsIgnoreCase("AS") &&
                !call.getOperandList().isEmpty()) {
            SqlNode firstOperand = call.getOperandList().get(0);
            if (firstOperand instanceof SqlIdentifier identifier) {
                return identifier;
            }
        }
        return null;
    }

    private void addSourceTable(SqlNode node, SqlNodeVisitor.Result result) {
        log.debug("addSourceTable: Processing node type: {}", node.getClass().getSimpleName());
        if (node instanceof SqlIdentifier id) {
            // Skip CTEs from being added as source tables
            String tableName = getTableName(id);
            if (tableName == null /*|| !result.getContext().getCteAliases().contains(tableName)*/) {
                result.getSourceTables().add(id);
                log.debug("addSourceTable: Added identifier to sourceTables: {}", String.join(".", id.names));
            }
        } else if (node instanceof SqlBasicCall call &&
                call.getOperator().getName().equalsIgnoreCase("AS") &&
                !call.getOperandList().isEmpty()) {
            SqlNode firstOperand = call.getOperandList().get(0);
            log.debug("addSourceTable: Node is AS call, first operand type: {}", firstOperand.getClass().getSimpleName());
            if (firstOperand instanceof SqlIdentifier tableId) {
                // Skip CTEs from being added as source tables
                String tableName = getTableName(tableId);
                if (tableName == null /*|| !result.getContext().getCteAliases().contains(tableName)*/) {
                    result.getSourceTables().add(tableId);
                    log.debug("addSourceTable: Added table from AS to sourceTables: {}", String.join(".", tableId.names));
                }
            } else if (firstOperand instanceof SqlSelect) {
                // Handle subquery with alias: recursively extract tables from the subquery
                log.debug("addSourceTable: AS contains subquery, recursively extracting tables");
                SqlNodeVisitor visitor = new SqlNodeVisitor(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
                SqlNodeVisitor.Result subqueryResult = firstOperand.accept(visitor);
                if (subqueryResult != null) {
                    log.debug("addSourceTable: Subquery result has {} sourceTables", subqueryResult.getSourceTables().size());
                    SqlNodeVisitor.mergeResults(result, subqueryResult);
                }
            }
        } else if (node instanceof SqlJoin joinNode) {
            log.debug("addSourceTable: Node is JOIN, processing left and right");
            addSourceTable(joinNode.getLeft(), result);
            addSourceTable(joinNode.getRight(), result);
        } else if (node instanceof SqlSelect subquery) {
            // Handle subquery directly: recursively extract tables from the subquery
            log.debug("addSourceTable: Node is direct subquery, recursively extracting tables");
            SqlNodeVisitor visitor = new SqlNodeVisitor(userName, defaultDatabase, defaultSchema, relationshipExtractor, entityCatalog);
            SqlNodeVisitor.Result subqueryResult = subquery.accept(visitor);
            if (subqueryResult != null) {
                log.debug("addSourceTable: Direct subquery result has {} sourceTables", subqueryResult.getSourceTables().size());
                SqlNodeVisitor.mergeResults(result, subqueryResult);
            }
        }
    }

    private String getTableName(SqlIdentifier id) {
        if (id instanceof com.calcite_new.sql.SqlTableIdentifier tableId) {
            return tableId.getTableName();
        } else if (id instanceof com.calcite_new.sql.SqlViewIdentifier viewId) {
            return viewId.getViewName();
        } else if (id.names != null && !id.names.isEmpty()) {
            return id.names.get(id.names.size() - 1);
        }
        return null;
    }

    private boolean checkCaseExpressionForInsensitiveComparison(SqlCase caseCall) {
        for (SqlNode operand : caseCall.getWhenOperands()) {
            if (operand instanceof SqlBasicCall) {
                if (SqlConditionUtils.hasCaseInsensitiveComparison(operand)) {
                    return true;
                }
            }
        }
        return false;
    }

    private void registerTablesForPartitionDetection(SqlNodeVisitor.Result result) {
        if (result == null || result.getSourceTables() == null) {
            return;
        }
        for (SqlIdentifier tableId : result.getSourceTables()) {
            partitionDetector.registerTable(tableId);
        }
    }

    /**
     * Computes a map from table name (normalized) to occurrence number.
     * The occurrence number represents the order in which the table appears in the query.
     * 
     * @param sourceTables The list of source tables from the query (preserves insertion order)
     * @return Map from normalized table name to occurrence number (1-based)
     */
    private Map<String, Integer> computeTableOccurrenceMap(List<SqlIdentifier> sourceTables) {
        return computeTableOccurrenceMapWithFirstOccurrence(sourceTables, 0);
    }

    /**
     * Computes a map from table name (normalized) to occurrence number.
     *
     * @param sourceTables The list of source tables from the query (preserves insertion order)
     * @param mainQueryTableCount The number of tables from the main query (before subqueries)
     * @return Map from normalized table name to occurrence number
     */
    private Map<String, Integer> computeTableOccurrenceMapWithFirstOccurrence(List<SqlIdentifier> sourceTables, int mainQueryTableCount) {
        Map<String, Integer> occurrenceMap = new HashMap<>();
        if (sourceTables == null || sourceTables.isEmpty()) {
            return occurrenceMap;
        }

        // Track first occurrence per table name
        Map<String, Integer> firstOccurrenceMap = new HashMap<>();
        // Track current occurrence count per table
        Map<String, Integer> tableCountMap = new HashMap<>();
        
        for (int i = 0; i < sourceTables.size(); i++) {
            SqlIdentifier tableId = sourceTables.get(i);
            String tableName = getTableNameFromEntity(tableId);

            if (tableName == null) {
                tableName = getTableName(tableId);
                if (tableName != null && (tableId instanceof com.calcite_new.sql.SqlTableIdentifier || 
                                          tableId instanceof com.calcite_new.sql.SqlViewIdentifier)) {
                    String entityTableName = getTableNameFromEntity(tableId);
                    if (entityTableName != null) {
                        tableName = entityTableName;
                    }
                }
            }
            
            if (tableName != null) {
                String normalizedTableName = tableName.toLowerCase();
                int currentCount = tableCountMap.getOrDefault(normalizedTableName, 0) + 1;
                tableCountMap.put(normalizedTableName, currentCount);

                if (!firstOccurrenceMap.containsKey(normalizedTableName)) {
                    firstOccurrenceMap.put(normalizedTableName, 1);
                    occurrenceMap.put(normalizedTableName, 1);
                } else {
                    occurrenceMap.put(normalizedTableName, currentCount);
                }
            }
        }
        
        return occurrenceMap;
    }

    private List<ColumnInfo> extractColumnsExcludingSubqueries(SqlNode whereExpr, ClauseType clauseType, String product, Map<String, Integer> occurrenceMap) {
        List<ColumnInfo> columnInfos = new ArrayList<>();
        extractColumnsExcludingSubqueriesRecursive(whereExpr, clauseType, product, columnInfos, occurrenceMap);
        return columnInfos;
    }
    private void extractColumnsExcludingSubqueriesRecursive(SqlNode node, ClauseType clauseType, String product, List<ColumnInfo> columnInfos, Map<String, Integer> occurrenceMap) {
        if (node == null) {
            return;
        }

        if (node instanceof SqlSelect) {
            return;
        }

        if (node instanceof com.calcite_new.sql.SqlColumnIdentifier ||
            (node instanceof SqlIdentifier id &&
             !(id instanceof com.calcite_new.sql.SqlTableIdentifier) &&
             !(id instanceof com.calcite_new.sql.SqlViewIdentifier))) {
            List<ColumnInfo> extracted = ColumnExtractor.extractColumns(node, clauseType, product, occurrenceMap);
            for (ColumnInfo ci : extracted) {
                if (ci != null && !columnInfos.contains(ci)) {
                    columnInfos.add(ci);
                }
            }
        }
        else if (node instanceof SqlCall call) {
            for (SqlNode operand : call.getOperandList()) {
                extractColumnsExcludingSubqueriesRecursive(operand, clauseType, product, columnInfos, occurrenceMap);
            }
        }
        else if (node instanceof SqlNodeList nodeList) {
            for (SqlNode item : nodeList.getList()) {
                extractColumnsExcludingSubqueriesRecursive(item, clauseType, product, columnInfos, occurrenceMap);
            }
        }
    }

    private List<ColumnInfo> extractColumnsFromSubqueries(SqlNode whereExpr, ClauseType clauseType, String product, Map<String, Integer> occurrenceMap, int mainQueryTableCount) {
        List<ColumnInfo> columnInfos = new ArrayList<>();
        extractColumnsFromSubqueriesRecursive(whereExpr, clauseType, product, columnInfos, occurrenceMap, mainQueryTableCount);
        return columnInfos;
    }
    private void extractColumnsFromSubqueriesRecursive(SqlNode node, ClauseType clauseType, String product, List<ColumnInfo> columnInfos, Map<String, Integer> occurrenceMap, int mainQueryTableCount) {
        if (node == null) {
            return;
        }

        if (node instanceof SqlSelect subquery) {
            if (subquery.getWhere() != null) {
                List<ColumnInfo> subqueryColumns = ColumnExtractor.extractColumns(subquery.getWhere(), clauseType, product, occurrenceMap);

                List<ColumnInfo> fixedColumns = new ArrayList<>();
                for (ColumnInfo ci : subqueryColumns) {
                    ColumnInfo fixed = ci;
                    if (ci.getTableName() == null && subquery.getFrom() != null) {
                        String inferredTableName = inferTableNameFromFromClause(subquery.getFrom());
                        if (inferredTableName != null && occurrenceMap != null) {
                            String normalizedTableName = inferredTableName.toLowerCase();
                            Integer mapOccurrence = occurrenceMap.get(normalizedTableName);
                            if (mapOccurrence != null) {
                                fixed = ColumnInfo.builder()
                                        .product(ci.getProduct())
                                        .database(ci.getDatabase())
                                        .schema(ci.getSchema())
                                        .tableName(inferredTableName)
                                        .columnName(ci.getColumnName())
                                        .clause(ci.getClause())
                                        .occurrence(mapOccurrence)
                                        .build();
                            }
                        }
                    }
                    else if (ci.getTableName() != null && occurrenceMap != null) {
                        String normalizedTableName = ci.getTableName().toLowerCase();
                        Integer mapOccurrence = occurrenceMap.get(normalizedTableName);
                        if (mapOccurrence != null && !mapOccurrence.equals(ci.getOccurrence())) {
                            fixed = ColumnInfo.builder()
                                    .product(ci.getProduct())
                                    .database(ci.getDatabase())
                                    .schema(ci.getSchema())
                                    .tableName(ci.getTableName())
                                    .columnName(ci.getColumnName())
                                    .clause(ci.getClause())
                                    .occurrence(mapOccurrence)
                                    .build();
                        }
                    }
                    fixedColumns.add(fixed);
                }
                columnInfos.addAll(fixedColumns);
            }
            return;
        }

        if (node instanceof SqlBasicCall call) {
            SqlOperator operator = call.getOperator();
            if (operator != null) {
                org.apache.calcite.sql.SqlKind kind = operator.getKind();
                String opName = operator.getName();

                if (kind == org.apache.calcite.sql.SqlKind.IN || kind == org.apache.calcite.sql.SqlKind.NOT_IN ||
                    "IN".equalsIgnoreCase(opName) || "NOT IN".equalsIgnoreCase(opName)) {
                    List<SqlNode> operands = call.getOperandList();
                    if (operands.size() >= 2) {
                        SqlNode rhs = operands.get(1);
                        if (rhs instanceof SqlSelect) {
                            extractColumnsFromSubqueriesRecursive(rhs, clauseType, product, columnInfos, occurrenceMap, mainQueryTableCount);
                            return;
                        }
                    }
                }
                else if (kind == org.apache.calcite.sql.SqlKind.EXISTS || "EXISTS".equalsIgnoreCase(opName)) {
                    List<SqlNode> operands = call.getOperandList();
                    if (!operands.isEmpty()) {
                        SqlNode operand = operands.get(0);
                        if (operand instanceof SqlSelect) {
                            extractColumnsFromSubqueriesRecursive(operand, clauseType, product, columnInfos, occurrenceMap, mainQueryTableCount);
                            return;
                        }
                    }
                }
            }

            for (SqlNode operand : call.getOperandList()) {
                extractColumnsFromSubqueriesRecursive(operand, clauseType, product, columnInfos, occurrenceMap, mainQueryTableCount);
            }
        }
        else if (node instanceof SqlNodeList nodeList) {
            for (SqlNode item : nodeList.getList()) {
                extractColumnsFromSubqueriesRecursive(item, clauseType, product, columnInfos, occurrenceMap, mainQueryTableCount);
            }
        }
    }

    /**
     * Extracts the table name from the entity's normalized name.
     * This matches what ColumnExtractor uses, ensuring proper occurrence tracking.
     */
    private String getTableNameFromEntity(SqlIdentifier id) {
        if (id instanceof com.calcite_new.sql.SqlTableIdentifier tableId) {
            var entity = tableId.getEntity();
            if (entity != null) {
                return entity.getName().getNormalizedName();
            }
        } else if (id instanceof com.calcite_new.sql.SqlViewIdentifier viewId) {
            var entity = viewId.getEntity();
            if (entity != null) {
                return entity.getName().getNormalizedName();
            }
        }
        return null;
    }

    /**
     * Infers table name from a FROM clause node.
     * Used when column references don't have table qualifiers.
     */
    private String inferTableNameFromFromClause(SqlNode from) {
        if (from instanceof SqlIdentifier id) {
            return getTableNameFromEntity(id);
        } else if (from instanceof SqlBasicCall call &&
                   call.getOperator() != null &&
                   call.getOperator().getName().equalsIgnoreCase("AS") &&
                   !call.getOperandList().isEmpty()) {
            SqlNode tableNode = call.getOperandList().get(0);
            if (tableNode instanceof SqlIdentifier) {
                return getTableNameFromEntity((SqlIdentifier) tableNode);
            }
        }
        return null;
    }

    private List<String> extractJoinTableNames(SqlJoin join) {
        List<String> tableNames = new ArrayList<>();
        addTableName(join.getLeft(), tableNames);
        addTableName(join.getRight(), tableNames);
        return tableNames;
    }

    private void addTableName(SqlNode node, List<String> tableNames) {
        if (node instanceof SqlIdentifier id) {
            String fullQualifiedName = String.join(".", id.names);
            if (!tableNames.contains(fullQualifiedName)) {
                tableNames.add(fullQualifiedName);
            }
        } else if (node instanceof SqlBasicCall call &&
                call.getOperator().getName().equalsIgnoreCase("AS")) {
            if (call.getOperandList().get(0) instanceof SqlIdentifier tableId) {
                String fullQualifiedName = String.join(".", tableId.names);
                if (!tableNames.contains(fullQualifiedName)) {
                    tableNames.add(fullQualifiedName);
                }
            }
        } else if (node instanceof SqlJoin joinNode) {
            addTableName(joinNode.getLeft(), tableNames);
            addTableName(joinNode.getRight(), tableNames);
        }
    }
