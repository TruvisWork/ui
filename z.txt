import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
from datetime import datetime, timedelta
import numpy as np

# Load your data
# df = pd.read_csv('your_data.csv')

# Sample data structure - replace with your actual data
data = {
    'start_time': ['2025-11-21 10:00:00', '2025-11-21 10:05:00', '2025-11-21 10:10:00'],
    'end_time': ['2025-11-21 10:30:00', '2025-11-21 10:15:00', '2025-11-21 10:40:00'],
    'slot_seconds': [3000, 1500, 5000],
    'query_type': ['procs', 'user_fired', 'sa_non_proc']
}

df = pd.DataFrame(data)
print(f"Processing {len(df)} queries...")

# Convert to datetime
df['start_time'] = pd.to_datetime(df['start_time'])
df['end_time'] = pd.to_datetime(df['end_time'])

# Calculate duration and instantaneous slot usage
df['duration_seconds'] = (df['end_time'] - df['start_time']).dt.total_seconds()
df['duration_seconds'] = df['duration_seconds'].replace(0, 1)  # Avoid division by zero

# CORRECT: This gives us the average concurrent slots during the query's execution
# If a query consumed 1000 slot_seconds over 10 seconds, it used 100 slots on average
df['slots_per_second'] = df['slot_seconds'] / df['duration_seconds']

print(f"Slot usage per query range: {df['slots_per_second'].min():.2f} to {df['slots_per_second'].max():.2f}")


def calculate_concurrent_slots_correct(df, bucket_size='15min'):
    """
    Calculate TRUE concurrent slot usage at each time point.
    
    The key insight: at any moment in time, total slots = sum of all running queries' slot rates
    """
    print(f"Using {bucket_size} time buckets...")
    
    # Get time range
    min_time = df['start_time'].min()
    max_time = df['end_time'].max()
    print(f"Time range: {min_time} to {max_time}")
    
    # Create time buckets - these are our sampling points
    time_range = pd.date_range(
        start=min_time.floor(bucket_size),
        end=max_time.ceil(bucket_size),
        freq=bucket_size
    )
    
    results = []
    
    # For each time point, find all queries running and sum their slot rates
    for i, bucket_start in enumerate(time_range[:-1]):
        bucket_end = time_range[i + 1]
        bucket_mid = bucket_start + (bucket_end - bucket_start) / 2
        
        # Find all queries running at this time point
        # A query is "running" if bucket overlaps with [start_time, end_time]
        running_queries = df[
            (df['start_time'] < bucket_end) & 
            (df['end_time'] > bucket_start)
        ]
        
        # CORRECT: Just sum the instantaneous slot rates
        # No weighting needed - if a query is active, it contributes its full rate
        slots_by_type = {
            'procs': running_queries[running_queries['query_type'] == 'procs']['slots_per_second'].sum(),
            'user_fired': running_queries[running_queries['query_type'] == 'user_fired']['slots_per_second'].sum(),
            'sa_non_proc': running_queries[running_queries['query_type'] == 'sa_non_proc']['slots_per_second'].sum()
        }
        
        results.append({
            'time': bucket_mid,
            'procs': slots_by_type['procs'],
            'user_fired': slots_by_type['user_fired'],
            'sa_non_proc': slots_by_type['sa_non_proc'],
            'total': sum(slots_by_type.values()),
            'query_count': len(running_queries)
        })
    
    return pd.DataFrame(results)


print("\nCalculating concurrent slot usage...")

# Calculate with different granularities
timeline_15min = calculate_concurrent_slots_correct(df, '15min')
timeline_4h = calculate_concurrent_slots_correct(df, '4h')

timeline_df = timeline_15min

print(f"\nGenerated {len(timeline_df)} time points")
print(f"Concurrent slots range: {timeline_df['total'].min():.2f} to {timeline_df['total'].max():.2f}")
print(f"Peak concurrent queries: {timeline_df['query_count'].max()}")

# Create visualization
fig, ax = plt.subplots(1, 1, figsize=(20, 8))

colors = {
    'procs': '#e74c3c',
    'user_fired': '#2ecc71', 
    'sa_non_proc': '#3498db'
}

# Plot stacked area chart
ax.fill_between(timeline_df['time'], 0, timeline_df['procs'],
                label='procs', color=colors['procs'], alpha=0.7)
ax.fill_between(timeline_df['time'], timeline_df['procs'],
                timeline_df['procs'] + timeline_df['sa_non_proc'],
                label='sa_non_proc', color=colors['sa_non_proc'], alpha=0.7)
ax.fill_between(timeline_df['time'], 
                timeline_df['procs'] + timeline_df['sa_non_proc'],
                timeline_df['total'],
                label='user_fired', color=colors['user_fired'], alpha=0.7)

# Overlay 4-hour average
ax.plot(timeline_4h['time'], timeline_4h['total'],
        color='black', linewidth=3, label='4-Hour Average', alpha=0.8)

# Find and mark peak
peak_idx = timeline_df['total'].idxmax()
peak = timeline_df.loc[peak_idx]

ax.scatter([peak['time']], [peak['total']], 
          color='red', s=200, zorder=5, marker='o', 
          edgecolors='black', linewidths=2)

ax.annotate(
    f"PEAK: {peak['total']:.1f} slots\n{peak['time'].strftime('%m/%d %H:%M')}\n({int(peak['query_count'])} concurrent queries)",
    xy=(peak['time'], peak['total']),
    xytext=(30, 30), textcoords='offset points',
    bbox=dict(boxstyle='round,pad=0.8', fc='yellow', alpha=0.9),
    arrowprops=dict(arrowstyle='->', connectionstyle='arc3,rad=0', lw=2),
    fontsize=11, fontweight='bold'
)

# Better x-axis formatting
days_range = (timeline_df['time'].max() - timeline_df['time'].min()).days

if days_range <= 2:
    major_interval = 4  # Every 4 hours
    date_fmt = '%m/%d %H:%M'
elif days_range <= 7:
    major_interval = 12  # Every 12 hours
    date_fmt = '%m/%d %H:%M'
else:
    major_interval = 24  # Every day
    date_fmt = '%m/%d'

ax.xaxis.set_major_locator(mdates.HourLocator(interval=major_interval))
ax.xaxis.set_major_formatter(mdates.DateFormatter(date_fmt))

# Labels and styling
ax.set_xlabel('Time (MM/DD HH:MM)', fontsize=14, fontweight='bold')
ax.set_ylabel('Concurrent Slot Usage', fontsize=14, fontweight='bold')
ax.set_title(
    f'BigQuery Concurrent Slot Usage - 30 Days Analysis\n({len(df)} queries, 15-min resolution with 4-hour moving average)',
    fontsize=16, fontweight='bold'
)

ax.legend(loc='upper left', fontsize=11)
ax.grid(True, alpha=0.3)

# Rotate labels
plt.setp(ax.xaxis.get_majorticklabels(), rotation=45, ha='right')

plt.tight_layout()

print("\nSaving figure...")
plt.savefig('bigquery_slot_usage_analysis.png', dpi=150, bbox_inches='tight')
print("Done! Figure saved.")
plt.show()
