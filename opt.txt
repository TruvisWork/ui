INSERT INTO AMH_FZ_REPORT_MARTS_TABLES_DEV.Payment_Mart
(
  report_name, create_timestamp, action_on_alert_timestamp, result_on_alert_timestamp,
  entity, portfolio, channel, class, final_portfolio, statemachineID, final_state,
  payment_source, transaction_status, sender_transaction_currency,
  number_of_payment_customers, sum_of_transaction_amount_usd,
  sum_of_transaction_original_amount, count_of_transactions, count_of_alerts,
  sum_of_transaction_amount_by_alerts, number_of_alerted_customers,
  sum_of_transaction_original_amount_by_alerts, sum_of_transaction_amount_gbp,
  sum_of_transaction_amount_gbp_by_alerts, lob, load_datetime
)

-- 🧱 1. Base Event Arrival - EARLY CHANNEL FILTER (Major Scan Reduction)
WITH base_arrival AS (
  SELECT
    id,
    timestamp,
    updatedTimestamp,
    alert
  FROM (
    SELECT
      id,
      timestamp,
      updatedTimestamp,
      alert,
      ROW_NUMBER() OVER (
        PARTITION BY id.identifier
        ORDER BY TIMESTAMP(timestamp) ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_arrival
    WHERE
      LOWER(id.channelId) = 'transfers'  -- ⚡ EARLY FILTER - Massive scan reduction
      AND updatedTimestamp >= '2025-09-22 00:00:00+00'
      AND TIMESTAMP_MILLIS(id.timestamp) >= TIMESTAMP('2025-09-22 00:00:00+00')
      AND TIMESTAMP_MILLIS(id.timestamp) < TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 2. Single Event Store Scan - Consolidated for all uses
, base_event_store_full AS (
  SELECT
    lifecycle_id,
    event_type,
    channel_type,
    source,
    sender_transaction_currency,
    sender_transaction_amount_dbl,
    customer_id,
    customer_type,
    customer_id_number,
    payment_message_source,
    segment_channel_type,
    fdz_channel,
    entity_type,
    payment_revision_code,
    tpp_payee_payer,
    channel_name,
    sender_transaction_type,
    outcomes_and_scores,
    rules_triggered,
    event_occurred_at,
    bq_insert_timestamp
  FROM AMH_FZ_FDR_DEV_SIT.event_store
  WHERE
    LOWER(event_type) IN ('transfer_initiation', 'feedback')
    AND bq_insert_timestamp >= '2025-09-22 00:00:00+00'
)

-- 🧱 3. Event Store Main (with decision filter)
, base_event_store AS (
  SELECT
    lifecycle_id,
    event_type,
    channel_type,
    source,
    sender_transaction_currency,
    sender_transaction_amount_dbl,
    customer_id,
    customer_type,
    customer_id_number,
    payment_message_source,
    segment_channel_type,
    fdz_channel,
    entity_type,
    payment_revision_code,
    tpp_payee_payer,
    channel_name,
    sender_transaction_type,
    outcomes_and_scores
  FROM (
    SELECT *,
      ROW_NUMBER() OVER (
        PARTITION BY lifecycle_id
        ORDER BY event_occurred_at ASC
      ) AS rn
    FROM base_event_store_full
    WHERE
      JSON_EXTRACT_SCALAR(outcomes_and_scores,"$.decision.outcomeDecision")
          IN ('approve','decline','review')
  )
  WHERE rn = 1
)

-- 🧱 4. Alerts
, base_arrival_alert AS (
  SELECT iden
  FROM (
    SELECT
      id.identifier as iden,
      ROW_NUMBER() OVER (
        PARTITION BY id.identifier
        ORDER BY TIMESTAMP(timestamp) ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_arrival
    WHERE
      LOWER(id.channelId) = 'transfers'  -- ⚡ EARLY FILTER
      AND alert = TRUE
      AND updatedTimestamp >= '2025-09-22 00:00:00+00'
      AND TIMESTAMP_MILLIS(id.timestamp) >= TIMESTAMP('2025-09-22 00:00:00+00')
      AND TIMESTAMP_MILLIS(id.timestamp) < TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 5. Event Store for Alerts (transfer_initiation only)
, base_event_store_alert AS (
  SELECT
    lifecycle_id,
    sender_transaction_amount_dbl,
    customer_id,
    customer_type,
    customer_id_number
  FROM (
    SELECT *,
      ROW_NUMBER() OVER (
        PARTITION BY lifecycle_id
        ORDER BY bq_insert_timestamp ASC
      ) AS rn
    FROM base_event_store_full
    WHERE LOWER(event_type) = 'transfer_initiation'
  )
  WHERE rn = 1
)

-- 🧱 6. State Updates
, base_state_updates AS (
  SELECT
    identifier, statemachineid, state_id, updatedAt
  FROM (
    SELECT
      identifier, statemachineid, state.id AS state_id, updatedAt,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_state_updates, UNNEST(ids)
    WHERE
      LOWER(channelId) = "transfers"  -- ⚡ EARLY FILTER
      AND updatedTimestamp >= '2025-09-22 00:00:00+00'
      AND LOWER(state.id) NOT IN ("closed")
      AND LOWER(statemachineid) NOT IN (
        'breach_status','decision','status_digital_activity',
        'status_transfers','transfer_status','operational_status')
  )
  WHERE rn = 1
)

-- 🧱 7. State Status
, base_state_status AS (
  SELECT identifier, state_id
  FROM (
    SELECT identifier, state.id AS state_id,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_state_updates, UNNEST(ids)
    WHERE
      LOWER(stateMachineId) = 'status'
      AND updatedTimestamp >= '2025-09-22 00:00:00+00'
      AND TIMESTAMP_MILLIS(updatedAt) >= '2025-09-22 00:00:00+00'
      AND TIMESTAMP_MILLIS(updatedAt) < '2025-10-06 16:00:00+00'
  )
  WHERE rn = 1
)

-- 🧱 8. Assignee Update
, base_assignee_update AS (
  SELECT identifier, updatedAt
  FROM (
    SELECT identifier, updatedAt,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_assignee_update, UNNEST(ids)
    WHERE updatedTimestamp >= '2025-09-22 00:00:00+00'
  )
  WHERE rn = 1
)

-- 🧱 9. Queue Change
, base_queue_changed AS (
  SELECT identifier, timestamp
  FROM (
    SELECT identifier, cm_event_queue_changed.timestamp,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY TIMESTAMP(cm_event_queue_changed.timestamp) DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_queue_changed, UNNEST(ids)
    WHERE updatedTimestamp >= '2025-09-22 00:00:00+00'
  )
  WHERE rn = 1
)

-- 🧱 10. Rules - Restored original logic with optimization
, rules AS (
  SELECT
    es_rule.lifecycle_id,
    STRING_AGG(wr.name) AS rule_metadata_names
  FROM (
    SELECT
      lifecycle_id,
      TRIM(rules_split) AS rules_triggered
    FROM (
      SELECT
        ca.id.identifier,
        es.lifecycle_id,
        es.rules_triggered
      FROM base_arrival ca
      INNER JOIN base_event_store_full es
        ON ca.id.identifier = es.lifecycle_id
    ) event_store_subset,
    UNNEST(SPLIT(rules_triggered, ';')) AS rules_split
  ) es_rule
  INNER JOIN AMH_FZ_FDR_DEV_SIT.workflow_rules_vw wr
    ON es_rule.rules_triggered = wr.id
  GROUP BY es_rule.lifecycle_id
)

-- 🔟 Final aggregation
SELECT
  "Payment MI" AS report_name,
  TIMESTAMP_TRUNC(TIMESTAMP_ADD(TIMESTAMP_MILLIS(ca.id.timestamp), INTERVAL 8 HOUR), DAY) AS create_timestamp,
  NULLIF(
    TIMESTAMP_TRUNC(
      GREATEST(
        COALESCE(TIMESTAMP_MILLIS(asu.updatedAt), TIMESTAMP('1970-01-01 00:00:00 UTC')),
        COALESCE(TIMESTAMP_MILLIS(su.updatedAt), TIMESTAMP('1970-01-01 00:00:00 UTC')),
        COALESCE(TIMESTAMP(ca.timestamp), TIMESTAMP('1970-01-01 00:00:00 UTC')),
        COALESCE(TIMESTAMP(q.timestamp), TIMESTAMP('1970-01-01 00:00:00 UTC'))
      ),
      HOUR
    ),
    TIMESTAMP('1970-01-01 00:00:00')
  ) AS action_on_alert_timestamp,
  CASE WHEN LOWER(su.state_id) IN ('genuine','fraud')
       THEN TIMESTAMP_TRUNC(TIMESTAMP_MILLIS(su.updatedAt), HOUR)
  END AS result_on_alert_timestamp,
  ca.id.payload.schema.customer_portfolio_region AS entity,
  ca.id.payload.schema.customer_portfolio_country AS portfolio,
  CASE
    WHEN LOWER(es.channel_type) = 'c' THEN 'Payment Card at Card Reader Terminal (including online purchase and ATM)'
    WHEN LOWER(es.channel_type) = 'd' THEN 'Payment Card or Number with Online Details and Device Fingerprint Information'
    WHEN LOWER(es.channel_type) = 'e' THEN 'Payment Card or Number with Online Details'
    WHEN LOWER(es.channel_type) = 'o' THEN 'Online Banking (internet, mobile phone)'
    WHEN LOWER(es.channel_type) = 'w' THEN 'Online Banking with device fingerprint information'
    WHEN LOWER(es.channel_type) = 'p' THEN 'Phone Banking'
    WHEN LOWER(es.channel_type) = 'h' THEN 'Self Bank Branch'
    WHEN LOWER(es.channel_type) = 'm' THEN 'Correspondence(for non-mon and check deposit)'
    WHEN LOWER(es.channel_type) = 'b' THEN 'Bank Processing(include bank initiated non-mon maintenance, ACH debit, EFT processing)'
    WHEN LOWER(es.channel_type) = 'f' THEN 'Financial Consultant'
    WHEN LOWER(es.channel_type) = 'r' THEN 'Other'
    WHEN LOWER(es.channel_type) = 's' THEN 'Merchant - Acquirer Processing with Device Fingerprint'
    WHEN LOWER(es.channel_type) = 't' THEN 'Merchant - Acquirer Processing'
    WHEN LOWER(es.channel_type) = 'u' THEN 'Unknown'
    WHEN LOWER(es.channel_type) = 'n' THEN 'NA'
    ELSE es.channel_type
  END AS channel,
  ca.id.payload.schema.customer_portfolio_class AS class,
  CASE
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'cd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'pers' THEN 'CIIOM WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'cd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'buss' THEN 'CIIOM CMB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'ms'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'pers' THEN 'M&S WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'fd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'pers' THEN 'First Direct WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'pers' THEN 'Red Brand WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'buss' THEN 'Red Brand CMB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country) = 'uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class) = 'priv' THEN 'Red Brand GPB'
  END AS final_portfolio,
  su.statemachineid AS statemachineID,
  st.state_id AS final_state,
  es.source AS payment_source,
  CASE
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision") = "approve" THEN "approved"
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision") = "review" THEN "review"
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision") = "decline" THEN "declined"
  END AS transaction_status,
  es.sender_transaction_currency AS sender_transaction_currency,
  COUNT(DISTINCT es.customer_id) AS number_of_payment_customers,
  SUM(es.sender_transaction_amount_dbl) AS sum_of_transaction_amount_usd,
  SUM(CAST(es.sender_transaction_amount_dbl AS NUMERIC) * 100) AS sum_of_transaction_original_amount,
  COUNT(DISTINCT ca.id.identifier) AS count_of_transactions,
  COUNT(DISTINCT aa.iden) AS count_of_alerts,
  SUM(es_a.sender_transaction_amount_dbl) AS sum_of_transaction_amount_by_alerts,
  COUNT(DISTINCT es_a.customer_id) AS number_of_alerted_customers,
  SUM(CAST(es_a.sender_transaction_amount_dbl AS NUMERIC) * 100) AS sum_of_transaction_original_amount_by_alerts,
  SUM(es.sender_transaction_amount_dbl) AS sum_of_transaction_amount_gbp,
  SUM(es_a.sender_transaction_amount_dbl) AS sum_of_transaction_amount_gbp_by_alerts,
  CASE
    WHEN (LOWER(es.payment_message_source) IN ("hk_gpb_mob","hk_gpb_web")
          AND LOWER(es.segment_channel_type) = 'w'
          AND LOWER(es.fdz_channel) = 'transfers'
          AND LOWER(es.event_type) = 'transfer_initiation'
          AND (LOWER(es.entity_type) != 'b' OR es.entity_type IS NULL)
          AND LOWER(es.payment_revision_code) = 'o'
          AND LOWER(es.tpp_payee_payer) IN ('e', 'b', 'n', 'd'))
      OR (LOWER(es.channel_name) = 'l' AND LOWER(es.sender_transaction_type) = 'bt')
      THEN 'GPB'
    ELSE CASE
      WHEN LOWER(COALESCE(es.customer_type, es_a.customer_type)) = 'b'
           AND LOWER(SUBSTR(COALESCE(es.customer_id_number, es_a.customer_id_number), 1, 2)) NOT IN ('xw','xk','xc') 
           THEN "WSB"
      WHEN LOWER(COALESCE(es.customer_type, es_a.customer_type)) = 'i'
           OR (LOWER(COALESCE(es.customer_type, es_a.customer_type)) = 'b'
               AND LOWER(SUBSTR(COALESCE(es.customer_id_number, es_a.customer_id_number), 1, 2)) IN ('xw','xk','xc')) 
           THEN "WPB"
    END
  END AS lob,
  TIMESTAMP('2025-10-06 16:00:00+00') AS load_datetime
FROM base_arrival ca
LEFT JOIN base_event_store es       ON ca.id.identifier = es.lifecycle_id
LEFT JOIN base_arrival_alert aa     ON ca.id.identifier = aa.iden
LEFT JOIN base_event_store_alert es_a ON aa.iden = es_a.lifecycle_id
LEFT JOIN base_state_updates su     ON ca.id.identifier = su.identifier
LEFT JOIN base_state_status st      ON ca.id.identifier = st.identifier
LEFT JOIN base_assignee_update asu  ON ca.id.identifier = asu.identifier
LEFT JOIN base_queue_changed q      ON ca.id.identifier = q.identifier
LEFT JOIN rules r                   ON ca.id.identifier = r.lifecycle_id
GROUP BY
  create_timestamp, action_on_alert_timestamp, result_on_alert_timestamp,
  entity, portfolio, channel, class, final_portfolio, statemachineID,
  final_state, payment_source, transaction_status,
  sender_transaction_currency, lob;
