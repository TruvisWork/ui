SELECT
  COALESCE(a.report_name, b.report_name) AS report_name,
  COALESCE(a.create_timestamp, b.create_timestamp) AS create_timestamp,
  CASE WHEN TO_JSON_STRING(a) != TO_JSON_STRING(b) THEN 'DIFFERENT' END AS diff_flag,
  a AS test_row,
  b AS optimised_row
FROM test a
FULL OUTER JOIN optimised b
USING (
  report_name,
  create_timestamp,
  entity,
  portfolio,
  channel,
  class,
  final_portfolio,
  statemachineID,
  final_state,
  payment_source,
  transaction_status,
  sender_transaction_currency,
  lob
)
WHERE TO_JSON_STRING(a) != TO_JSON_STRING(b)
LIMIT 50;

-- 🧱 1. Base Event Arrival (deduped, column-pruned)
WITH base_arrival AS (
  SELECT
    id,
    timestamp,
    updatedTimestamp,
    alert
  FROM (
    SELECT
      id,
      timestamp,
      updatedTimestamp,
      alert,
      ROW_NUMBER() OVER (
        PARTITION BY id.identifier
        ORDER BY TIMESTAMP(timestamp) ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_arrival
    WHERE
      updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                           AND TIMESTAMP('2025-10-06 16:00:00+00')
      AND TIMESTAMP_MILLIS(id.timestamp)
          BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
              AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 2. Event Store (deduped, column-pruned)
, base_event_store AS (
  SELECT
    lifecycle_id,
    event_type,
    channel_type,
    source,
    sender_transaction_currency,
    sender_transaction_amount_dbl,
    customer_id,
    customer_type,
    customer_id_number,
    payment_message_source,
    segment_channel_type,
    fdz_channel,
    entity_type,
    payment_revision_code,
    channel_name,
    sender_transaction_type,
    outcomes_and_scores,
    bq_insert_timestamp
  FROM (
    SELECT *,
      ROW_NUMBER() OVER (
        PARTITION BY lifecycle_id
        ORDER BY event_occurred_at ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.event_store
    WHERE
      LOWER(event_type) IN ('transfer_initiation', 'feedback')
      AND JSON_EXTRACT_SCALAR(outcomes_and_scores,"$.decision.outcomeDecision")
          IN ('approve','decline','review')
      AND bq_insert_timestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                                 AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 3. Alerts
, base_arrival_alert AS (
  SELECT id.identifier
  FROM (
    SELECT
      id.identifier,
      ROW_NUMBER() OVER (
        PARTITION BY id.identifier
        ORDER BY TIMESTAMP(timestamp) ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_arrival
    WHERE
      alert = TRUE
      AND updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                               AND TIMESTAMP('2025-10-06 16:00:00+00')
      AND TIMESTAMP_MILLIS(id.timestamp)
          BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
              AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 4. Event Store for Alerts
, base_event_store_alert AS (
  SELECT
    lifecycle_id,
    sender_transaction_amount_dbl,
    customer_id
  FROM (
    SELECT *,
      ROW_NUMBER() OVER (
        PARTITION BY lifecycle_id
        ORDER BY bq_insert_timestamp ASC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.event_store
    WHERE
      LOWER(event_type)='transfer_initiation'
      AND bq_insert_timestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                                 AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 5. State Updates
, base_state_updates AS (
  SELECT
    identifier, statemachineid, state.id, updatedAt
  FROM (
    SELECT
      identifier, statemachineid, state.id, updatedAt,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_state_updates, UNNEST(ids)
    WHERE
      updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                           AND TIMESTAMP('2025-10-06 16:00:00+00')
      AND LOWER(state.id) NOT IN ("closed")
      AND LOWER(channelId)="transfers"
      AND LOWER(statemachineid) NOT IN (
        'breach_status','decision','status_digital_activity',
        'status_transfers','transfer_status','operational_status')
  )
  WHERE rn = 1
)

-- 🧱 6. State Status
, base_state_status AS (
  SELECT identifier, state.id
  FROM (
    SELECT identifier, state.id,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_state_updates, UNNEST(ids)
    WHERE
      LOWER(stateMachineId)='status'
      AND updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                               AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 7. Assignee Update
, base_assignee_update AS (
  SELECT identifier, updatedAt
  FROM (
    SELECT identifier, updatedAt,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY updatedAt DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_assignee_update, UNNEST(ids)
    WHERE updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                               AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 8. Queue Change
, base_queue_changed AS (
  SELECT identifier, cm_event_queue_changed.timestamp
  FROM (
    SELECT identifier, cm_event_queue_changed.timestamp,
      ROW_NUMBER() OVER (
        PARTITION BY identifier
        ORDER BY TIMESTAMP(cm_event_queue_changed.timestamp) DESC
      ) AS rn
    FROM AMH_FZ_FDR_DEV_SIT.cm_event_queue_changed, UNNEST(ids)
    WHERE updatedTimestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                               AND TIMESTAMP('2025-10-06 16:00:00+00')
  )
  WHERE rn = 1
)

-- 🧱 9. Rules (renamed alias)
, rules AS (
  SELECT
    es_rule.lifecycle_id,
    STRING_AGG(wr.name) AS rule_metadata_names
  FROM (
    SELECT
      e.lifecycle_id,
      TRIM(rule_id) AS rule_id
    FROM AMH_FZ_FDR_DEV_SIT.event_store e,
      UNNEST(SPLIT(e.rules_triggered, ';')) AS rule_id
    WHERE
      LOWER(e.event_type) IN ('transfer_initiation', 'feedback')
      AND e.bq_insert_timestamp BETWEEN TIMESTAMP('2025-09-22 00:00:00+00')
                                   AND TIMESTAMP('2025-10-06 16:00:00+00')
  ) es_rule
  JOIN AMH_FZ_FDR_DEV_SIT.workflow_rules_vw wr
    ON es_rule.rule_id = wr.id
  GROUP BY es_rule.lifecycle_id
)

-- 🔟 Final aggregation
SELECT
  "Payment MI" AS report_name,
  TIMESTAMP_TRUNC(TIMESTAMP_ADD(TIMESTAMP_MILLIS(ca.id.timestamp), INTERVAL 8 HOUR), DAY) AS create_timestamp,
  TIMESTAMP_TRUNC(
    GREATEST(
      COALESCE(TIMESTAMP_MILLIS(asu.updatedAt), TIMESTAMP('1970-01-01')),
      COALESCE(TIMESTAMP_MILLIS(su.updatedAt), TIMESTAMP('1970-01-01')),
      COALESCE(TIMESTAMP(ca.timestamp), TIMESTAMP('1970-01-01')),
      COALESCE(TIMESTAMP(q.timestamp), TIMESTAMP('1970-01-01'))
    ),
    HOUR
  ) AS action_on_alert_timestamp,
  CASE WHEN LOWER(su.state.id) IN ('genuine','fraud')
       THEN TIMESTAMP_TRUNC(TIMESTAMP_MILLIS(su.updatedAt), HOUR)
  END AS result_on_alert_timestamp,
  ca.id.payload.schema.customer_portfolio_region AS entity,
  ca.id.payload.schema.customer_portfolio_country AS portfolio,
  es.channel_type AS channel,
  ca.id.payload.schema.customer_portfolio_class AS class,
  CASE
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='cd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='pers' THEN 'CIIOM WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='cd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='buss' THEN 'CIIOM CMB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='ms'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='pers' THEN 'M&S WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='fd'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='pers' THEN 'First Direct WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='pers' THEN 'Red Brand WPB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='buss' THEN 'Red Brand CMB'
    WHEN LOWER(ca.id.payload.schema.customer_portfolio_country)='uk'
         AND LOWER(ca.id.payload.schema.customer_portfolio_class)='priv' THEN 'Red Brand GPB'
  END AS final_portfolio,
  su.statemachineid AS statemachineID,
  st.state.id AS final_state,
  es.source AS payment_source,
  CASE
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision")="approve" THEN "approved"
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision")="review" THEN "review"
    WHEN JSON_EXTRACT_SCALAR(es.outcomes_and_scores,"$.decision.outcomeDecision")="decline" THEN "declined"
  END AS transaction_status,
  es.sender_transaction_currency AS sender_transaction_currency,
  COUNT(DISTINCT es.customer_id) AS number_of_payment_customers,
  SUM(es.sender_transaction_amount_dbl) AS sum_of_transaction_amount_usd,
  SUM(CAST(es.sender_transaction_amount_dbl AS NUMERIC)*100) AS sum_of_transaction_original_amount,
  COUNT(DISTINCT ca.id.identifier) AS count_of_transactions,
  COUNT(DISTINCT aa.identifier) AS count_of_alerts,
  SUM(es_a.sender_transaction_amount_dbl) AS sum_of_transaction_amount_by_alerts,
  COUNT(DISTINCT es_a.customer_id) AS number_of_alerted_customers,
  SUM(CAST(es_a.sender_transaction_amount_dbl AS NUMERIC)*100) AS sum_of_transaction_original_amount_by_alerts,
  SUM(es.sender_transaction_amount_dbl) AS sum_of_transaction_amount_gbp,
  SUM(es_a.sender_transaction_amount_dbl) AS sum_of_transaction_amount_gbp_by_alerts,
  CASE
    WHEN (LOWER(es.payment_message_source) IN ("hk_gpb_mob","hk_gpb_web")
          AND LOWER(es.segment_channel_type)='w'
          AND LOWER(es.fdz_channel)='transfers'
          AND LOWER(es.event_type)='transfer_initiation'
          AND (LOWER(es.entity_type)!='b' OR es.entity_type IS NULL)
          AND LOWER(es.payment_revision_code)='o')
      OR (LOWER(es.channel_name)='l' AND LOWER(es.sender_transaction_type)='bt')
      THEN 'GPB'
    ELSE CASE
      WHEN LOWER(es.customer_type)='b'
           AND LOWER(SUBSTR(es.customer_id_number,1,2)) NOT IN ('xw','xk','xc') THEN "WSB"
      WHEN LOWER(es.customer_type)='i'
           OR (LOWER(es.customer_type)='b'
               AND LOWER(SUBSTR(es.customer_id_number,1,2)) IN ('xw','xk','xc')) THEN "WPB"
    END
  END AS lob,
  TIMESTAMP('2025-10-06 16:00:00+00') AS load_datetime
FROM base_arrival ca
LEFT JOIN base_event_store es       ON ca.id.identifier = es.lifecycle_id
LEFT JOIN base_arrival_alert aa     ON ca.id.identifier = aa.identifier
LEFT JOIN base_event_store_alert es_a ON aa.identifier = es_a.lifecycle_id
LEFT JOIN base_state_updates su     ON ca.id.identifier = su.identifier
LEFT JOIN base_state_status st      ON ca.id.identifier = st.identifier
LEFT JOIN base_assignee_update asu  ON ca.id.identifier = asu.identifier
LEFT JOIN base_queue_changed q      ON ca.id.identifier = q.identifier
LEFT JOIN rules r                   ON ca.id.identifier = r.lifecycle_id
WHERE LOWER(ca.id.channelId)='transfers'
GROUP BY
  create_timestamp, action_on_alert_timestamp, result_on_alert_timestamp,
  entity, portfolio, channel, class, final_portfolio, statemachineID,
  final_state, payment_source, transaction_status,
  sender_transaction_currency, lob;
