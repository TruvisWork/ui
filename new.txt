Query 1: Time-series data for graphing (per-minute aggregation for readability)

WITH job_metadata AS (
  SELECT
    job_id,
    user_email,
    statement_type,
    CASE 
      WHEN user_email NOT LIKE '%gserviceaccount.com' THEN 'user_adhoc'
      WHEN user_email LIKE '%gserviceaccount.com' 
           AND statement_type IN ('SCRIPT', 'CALL') THEN 'procedures'
      WHEN user_email LIKE '%gserviceaccount.com' THEN 'scheduled_sa'
    END AS workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
    AND job_type = 'QUERY'
),

timeline_with_type AS (
  SELECT
    t.period_start,
    t.period_slot_ms / 1000.0 AS slots_per_second,  -- convert slot-ms to slots
    m.workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT t
  JOIN job_metadata m ON t.job_id = m.job_id
  WHERE t.period_start > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
)

SELECT
  TIMESTAMP_TRUNC(period_start, MINUTE) AS minute,
  workload_type,
  AVG(slots_per_second) AS avg_slots_per_sec,
  MAX(slots_per_second) AS max_slots_per_sec
FROM timeline_with_type
WHERE workload_type IS NOT NULL
GROUP BY 1, 2
ORDER BY 1, 2;

Query 2: Summary statistics (p50, p90, max) per workload type

WITH job_metadata AS (
  SELECT
    job_id,
    user_email,
    statement_type,
    CASE 
      WHEN user_email NOT LIKE '%gserviceaccount.com' THEN 'user_adhoc'
      WHEN user_email LIKE '%gserviceaccount.com' 
           AND statement_type IN ('SCRIPT', 'CALL') THEN 'procedures'
      WHEN user_email LIKE '%gserviceaccount.com' THEN 'scheduled_sa'
    END AS workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 DAY)
    AND job_type = 'QUERY'
),

timeline_with_type AS (
  SELECT
    t.period_start,
    t.period_slot_ms / 1000.0 AS slots_per_second,
    m.workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT t
  JOIN job_metadata m ON t.job_id = m.job_id
  WHERE t.period_start > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 DAY)
)

SELECT
  workload_type,
  ROUND(APPROX_QUANTILES(slots_per_second, 100)[OFFSET(50)], 2) AS p50_slots_per_sec,
  ROUND(APPROX_QUANTILES(slots_per_second, 100)[OFFSET(90)], 2) AS p90_slots_per_sec,
  ROUND(APPROX_QUANTILES(slots_per_second, 100)[OFFSET(99)], 2) AS p99_slots_per_sec,
  ROUND(MAX(slots_per_second), 2) AS max_slots_per_sec,
  COUNT(*) AS total_seconds_sampled
FROM timeline_with_type
WHERE workload_type IS NOT NULL
GROUP BY 1
ORDER BY 1;

Query 3: Hourly heatmap data (slots by hour-of-day for pattern analysis)

WITH job_metadata AS (
  SELECT
    job_id,
    user_email,
    statement_type,
    CASE 
      WHEN user_email NOT LIKE '%gserviceaccount.com' THEN 'user_adhoc'
      WHEN user_email LIKE '%gserviceaccount.com' 
           AND statement_type IN ('SCRIPT', 'CALL') THEN 'procedures'
      WHEN user_email LIKE '%gserviceaccount.com' THEN 'scheduled_sa'
    END AS workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 14 DAY)
    AND job_type = 'QUERY'
),

timeline_with_type AS (
  SELECT
    t.period_start,
    t.period_slot_ms / 1000.0 AS slots_per_second,
    m.workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT t
  JOIN job_metadata m ON t.job_id = m.job_id
  WHERE t.period_start > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 14 DAY)
)

SELECT
  workload_type,
  EXTRACT(HOUR FROM period_start) AS hour_of_day,
  ROUND(AVG(slots_per_second), 2) AS avg_slots_per_sec,
  ROUND(APPROX_QUANTILES(slots_per_second, 100)[OFFSET(90)], 2) AS p90_slots_per_sec,
  ROUND(MAX(slots_per_second), 2) AS max_slots_per_sec
FROM timeline_with_type
WHERE workload_type IS NOT NULL
GROUP BY 1, 2
ORDER BY 1, 2;


Query 4: Pivoted format for easier graphing (3 columns for 3 workload types)

WITH job_metadata AS (
  SELECT
    job_id,
    user_email,
    statement_type,
    CASE 
      WHEN user_email NOT LIKE '%gserviceaccount.com' THEN 'user_adhoc'
      WHEN user_email LIKE '%gserviceaccount.com' 
           AND statement_type IN ('SCRIPT', 'CALL') THEN 'procedures'
      WHEN user_email LIKE '%gserviceaccount.com' THEN 'scheduled_sa'
    END AS workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
    AND job_type = 'QUERY'
),

timeline_with_type AS (
  SELECT
    TIMESTAMP_TRUNC(t.period_start, MINUTE) AS minute,
    t.period_slot_ms / 1000.0 AS slots_per_second,
    m.workload_type
  FROM `region-us`.INFORMATION_SCHEMA.JOBS_TIMELINE_BY_PROJECT t
  JOIN job_metadata m ON t.job_id = m.job_id
  WHERE t.period_start > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 7 DAY)
)

SELECT
  minute,
  ROUND(SUM(CASE WHEN workload_type = 'user_adhoc' THEN slots_per_second ELSE 0 END), 2) AS user_adhoc_slots,
  ROUND(SUM(CASE WHEN workload_type = 'scheduled_sa' THEN slots_per_second ELSE 0 END), 2) AS scheduled_sa_slots,
  ROUND(SUM(CASE WHEN workload_type = 'procedures' THEN slots_per_second ELSE 0 END), 2) AS procedures_slots,
  ROUND(SUM(slots_per_second), 2) AS total_slots
FROM timeline_with_type
WHERE workload_type IS NOT NULL
GROUP BY 1
ORDER BY 1;
