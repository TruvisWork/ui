WITH slot_usage AS (
  SELECT
    TIMESTAMP_TRUNC(creation_time, HOUR) AS hour,
    user_email,
    CASE 
      WHEN user_email LIKE '%gserviceaccount.com' 
           AND statement_type = 'SCRIPT' THEN 'procedures'
      WHEN user_email LIKE '%gserviceaccount.com' THEN 'scheduled'
      ELSE 'adhoc'
    END AS workload_type,
    total_slot_ms
  FROM `region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time > TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 DAY)
    AND job_type = 'QUERY'
    AND state = 'DONE'
)
SELECT
  workload_type,
  EXTRACT(HOUR FROM hour) AS hour_of_day,
  APPROX_QUANTILES(total_slot_ms / 3600000, 100)[OFFSET(50)] AS p50_slots,
  APPROX_QUANTILES(total_slot_ms / 3600000, 100)[OFFSET(95)] AS p95_slots,
  MAX(total_slot_ms / 3600000) AS max_slots
FROM slot_usage
GROUP BY 1, 2
ORDER BY 1, 2;
